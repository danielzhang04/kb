\# scripts/prospecting/deploy_preflight.py
from __future__ import annotations
from dataclasses import dataclass
import argparse
import json
import subprocess
from pathlib import Path
from typing import Callable

@dataclass(frozen=True)
class Check:
    name: str
    ok: bool
    code: str

@dataclass
class Environment:
    root: Path
    merged: Callable[[],bool]
    gates: Callable[[],tuple[str,...]]
    version: Callable[[str],str]
    tailscale: Callable[[],bool]
    ssh_noop: Callable[[],bool]
    secret_paths: Callable[[],tuple[str,...]]

def check(env: Environment) -> tuple[Check,...]:
    expected={"python":"3.13.7","datasette":"0.65.1","playwright":"1.55.0"}
    merged_ok=env.merged()
    gate_set=env.gates()
    results=[
        Check("merged",merged_ok,"ok" if merged_ok else "branch_not_merged"),
        Check("gates",gate_set==("P1","P2","P3","P4","P5"),"recorded_gate_set"),
    ]
    for name,wanted in expected.items():
        actual=env.version(name)
        results.append(Check(name,actual==wanted,"version_match" if actual==wanted else "version_mismatch"))
    results.extend([
        Check("tailscale",env.tailscale(),"peer_reachable"),
        Check("desktop_ssh",env.ssh_noop(),"fixed_noop"),
        Check("tracked_secrets",not env.secret_paths(),"paths_only_scan"),
    ])
    return tuple(results)

def _run(argv,**kwargs):
    return subprocess.run(argv,**kwargs)

def production(root: Path,desktop_host: str,expected_merge_sha: str,runner=_run) -> Environment:
    def run(argv):
        return runner(argv,cwd=root,text=True,capture_output=True,check=False,shell=False)
    def merged():
        branch=run(["git","branch","--show-current"]).stdout.strip()
        head=run(["git","rev-parse","HEAD"]).stdout.strip()
        promoted=run(
            ["git","merge-base","--is-ancestor",expected_merge_sha,"HEAD"]
        ).returncode==0
        return branch=="ops" and head==expected_merge_sha and promoted
    def gates():
        names=[]
        for phase in ("P1","P2","P3","P4","P5"):
            result=run(["py","-3","-m","scripts.prospecting.gate","--phase",phase,"--verify-recorded"])
            try:
                value=json.loads(result.stdout)
            except json.JSONDecodeError:
                continue
            if result.returncode==0 and value.get("recorded")=="matched" and value.get("passed") is True:
                names.append(phase)
        return tuple(names)
    def version(name):
        argv={"python":["py","-3","--version"],"datasette":["datasette","--version"],
              "playwright":["py","-3","-m","playwright","--version"]}[name]
        return run(argv).stdout.strip().split()[-1]
    def tailscale():
        result=run(["tailscale","status","--json"])
        if result.returncode != 0:
            return False
        peers=json.loads(result.stdout).get("Peer",{}).values()
        return any(peer.get("HostName")==desktop_host and peer.get("Online") is True for peer in peers)
    def ssh_noop():
        return run(["ssh","--",desktop_host,"py","-3","-c","raise SystemExit(0)"]).returncode==0
    def secret_paths():
        result=run([
            "git","grep","-Il","-E",
            "(API_KEY|CLIENT_SECRET|ACCESS_TOKEN|REFRESH_TOKEN)[[:space:]]*=",
            "--",
        ],root)
        return tuple(line for line in result.stdout.splitlines() if line)
    return Environment(
        root,merged,gates,version,tailscale,ssh_noop,secret_paths
    )

def main(argv=None):
    parser=argparse.ArgumentParser()
    parser.add_argument("--desktop-host",required=True)
    parser.add_argument("--expected-merge-sha",required=True)
    parser.add_argument("--root",type=Path,default=Path.cwd())
    args=parser.parse_args(argv)
    results=check(production(args.root,args.desktop_host,args.expected_merge_sha))
    print(json.dumps(
        {"ready":all(item.ok for item in results),
         "checks":[{"name":i.name,"ok":i.ok,"code":i.code} for i in results]},
        sort_keys=True,separators=(",",":"),
    ))
    return 0 if all(item.ok for item in results) else 1

if __name__=="__main__":
    raise SystemExit(main())
