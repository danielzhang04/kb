Implement the tier-0 prospecting campaigner as a deterministic, desktop-local draft workflow: enroll only approved immutable revisions, schedule within policy, reconcile Gmail inbound state through the P1 executor, create correctly threaded Gmail drafts without ever requesting send, stop safely, and pass the numeric P4 gate in recorded/fake mode.

\## Architecture

P4 adds a campaigner package behind P1's SQLite repository and deterministic executor. The campaigner owns enrollment, delivery scheduling, inbound classification, suppression, labels, reply revisions, and draft release. It has no Gmail credential or raw Gmail tool. All four Gmail capabilities are typed `exec_request` operations handled by `gmail_adapter.py`; the adapter talks to the installed Gmail MCP only outside `KB_PROSPECTING_NO_NETWORK=1`, and otherwise uses a stateful fake/recorded transport.

Release is a bounded state machine. For every due delivery the executor serializes the Gmail thread, refreshes it, rechecks all stops, blocks, approval hashes, caps, and tier, then CAS-claims the delivery. That refresh plus CAS is the linearization point. P4 creates a draft and persists its deterministic RFC Message-ID and Gmail identifiers. An inbound arriving after the CAS is outside the prevention guarantee and is reconciled on the next inbound pass.

class FakeGmail:
    def draft_new(self, subject: str, body: str, rfc_message_id: str) -> FakeMessage:
        message_id, thread_id, history_id = self._ids()
        if self.messages_list(f"rfc822msgid:{rfc_message_id}"):
            raise ValueError("duplicate_rfc_message_id")
        message = FakeMessage(
            message_id, thread_id, rfc_message_id, subject, body,
            {"Message-ID": rfc_message_id, "Subject": subject},
            False, True, history_id,
        )
        self._threads[thread_id] = FakeThread(thread_id, subject, [message])
        self._record("draftAdded", message)
        return message

def validate_p4_inventory(manifest: Mapping[str, object]) -> tuple[str, ...]:
    declared = tuple(map(str, manifest.get("artifacts", ())))
    errors = [
        f"artifact_not_allowlisted:{path}"
        for path in declared if path not in P4_ARTIFACT_ALLOWLIST
    ]
    if len(declared) != len(set(declared)):
        errors.append("artifacts:duplicate")
    if set(manifest.get("fixtures", ())) != {
        "gmail-threads.json", "gmail-uncertain.json", "inbound-classes.json",
        "send-race-boundaries.json", "reply-drafts.json", "google-warning.json",
    }:
        errors.append("fixtures:inventory")
    return tuple(sorted(errors))


def collect_manifest_node_ids(root: Path, manifest: Mapping[str, object]) -> frozenset[str]:
    files = sorted({str(node).split("::", 1)[0] for node in manifest["tests"]})
    environment = {**os.environ, "KB_PROSPECTING_NO_NETWORK": "1"}
    completed = subprocess.run(
        [sys.executable, "-m", "pytest", "--collect-only", "-q", *files],
        cwd=root, env=environment, text=True, capture_output=True, check=False,
    )
    if completed.returncode != 0:
        raise RuntimeError("p4_collect_failed")
    return frozenset(
        line.strip() for line in completed.stdout.splitlines()
        if "::test_" in line and not line.startswith(" ")
    )


def validate_p4_fixtures(fixtures: Mapping[str, object]) -> tuple[str, ...]:

def _run_node(root: Path, node_id: str, repetitions: int) -> int:
    environment = {**os.environ, "KB_PROSPECTING_NO_NETWORK": "1"}
    completed_runs = 0
    for _ in range(repetitions):
        completed = subprocess.run(
            [sys.executable, "-m", "pytest", node_id, "-q"], cwd=root,
            env=environment, text=True, capture_output=True, check=False,
        )
        report = (completed.stdout + completed.stderr).casefold()
        if completed.returncode != 0 or any(
            marker in report for marker in (" warning", " skipped", " xfailed", " xpassed")
        ):
            raise RuntimeError("p4_measurement_failed")
        completed_runs += 1
    return completed_runs
