#!/usr/bin/env python3
"""Metadata-only inventory scanner for manifest-approved migration sources.

The scanner does not open regular-file contents, compute hashes, follow reparse
points, accept continuation cursors, overwrite reports, or scan sources absent
from manifest.json.
"""

from __future__ import annotations

import argparse
import json
import os
import re
import stat
import sys
import time
from collections import Counter
from dataclasses import dataclass
from datetime import datetime, timezone
from pathlib import Path
from typing import Any, TextIO

REPARSE_ATTRIBUTE = getattr(stat, "FILE_ATTRIBUTE_REPARSE_POINT", 0x400)
DIRECTORY_ATTRIBUTE = getattr(stat, "FILE_ATTRIBUTE_DIRECTORY", 0x10)

ATTRIBUTE_NAMES = (
    (0x0001, "readonly"),
    (0x0002, "hidden"),
    (0x0004, "system"),
    (0x0010, "directory"),
    (0x0020, "archive"),
    (0x0080, "normal"),
    (0x0100, "temporary"),
    (0x0200, "sparse"),
    (0x0400, "reparse_point"),
    (0x0800, "compressed"),
    (0x1000, "offline"),
    (0x2000, "not_content_indexed"),
    (0x4000, "encrypted"),
    (0x8000, "integrity_stream"),
    (0x10000, "virtual"),
    (0x20000, "no_scrub_data"),
    (0x80000, "pinned"),
    (0x100000, "unpinned"),
)

PROTECTED_NAMES = frozenset({".aws", ".ssh", ".gnupg"})
PREFIX_RE = re.compile(r"[A-Za-z0-9][A-Za-z0-9._-]{0,100}\Z")

TOCTOU_WARNING = (
    "This is an unlocked metadata observation, not a filesystem snapshot. "
    "A path can be replaced or modified between metadata checks and directory "
    "enumeration. The scanner rechecks directory identity immediately before "
    "enumeration, but Python stdlib does not provide an atomic Windows "
    "open-directory-without-reparse traversal primitive. Freeze and reconcile "
    "the affected source before recovery or cutover."
)


class InventoryRefusal(RuntimeError):
    """A fail-closed precondition or output-safety refusal."""


@dataclass(frozen=True)
class ManifestEntry:
    source: str
    destination: str
    alias: str
    status: str | None


@dataclass
class ScanState:
    entry_count: int = 0
    error_count: int = 0
    policy_skip_count: int = 0
    reparse_count: int = 0
    file_bytes: int = 0
    reparse_bytes: int = 0
    max_source_utf16_units: int = 0
    max_projected_utf16_units: int = 0
    enumeration_cutoff: bool = False
    stop_reason: str | None = None


def extended_path(path: str) -> str:
    """Return an absolute Windows extended-length path."""
    path = os.path.abspath(path)
    if path.startswith("\\\\?\\"):
        return path
    if path.startswith("\\\\"):
        return "\\\\?\\UNC\\" + path[2:]
    return "\\\\?\\" + path


def display_path(path: str) -> str:
    if path.startswith("\\\\?\\UNC\\"):
        return "\\\\" + path[8:]
    if path.startswith("\\\\?\\"):
        return path[4:]
    return path


def normalized(path: str) -> str:
    return os.path.normcase(os.path.normpath(os.path.abspath(path)))


def same_path(left: str, right: str) -> bool:
    return normalized(left) == normalized(right)


def is_within(path: str, parent: str) -> bool:
    try:
        return same_path(os.path.commonpath([normalized(path), normalized(parent)]), parent)
    except ValueError:
        return False


def utf16_units(value: str) -> int:
    return len(value.encode("utf-16-le", errors="surrogatepass")) // 2


def path_components(path: str) -> list[str]:
    _, tail = os.path.splitdrive(os.path.abspath(path))
    return [
        part.casefold()
        for part in tail.replace("/", "\\").split("\\")
        if part
    ]


def attributes_for(st: os.stat_result) -> int:
    return int(getattr(st, "st_file_attributes", 0))


def attribute_names(value: int) -> list[str]:
    return [name for mask, name in ATTRIBUTE_NAMES if value & mask]


def is_reparse(st: os.stat_result) -> bool:
    return bool(attributes_for(st) & REPARSE_ATTRIBUTE)


def safe_error(exc: OSError) -> dict[str, Any]:
    \# OSError text is metadata about the operation; no file content is read.
    return {
        "winerror": getattr(exc, "winerror", None),
        "errno": getattr(exc, "errno", None),
        "message": getattr(exc, "strerror", None) or exc.__class__.__name__,
    }


def lstat(path: str) -> os.stat_result:
    return os.stat(extended_path(path), follow_symlinks=False)


def lexical_exists(path: str) -> bool:
    return os.path.lexists(extended_path(path))


def assert_no_reparse_ancestors(path: str, require_leaf: bool = True) -> None:
    current = os.path.abspath(path)
    first = True

    while True:
        if first and not require_leaf and not lexical_exists(current):
            pass
        else:
            try:
                st = lstat(current)
            except OSError as exc:
                raise InventoryRefusal(
                    f"cannot inspect ancestor {current!r}: "
                    f"winerror={getattr(exc, 'winerror', None)!r}, "
                    f"errno={getattr(exc, 'errno', None)!r}"
                ) from exc
            if is_reparse(st):
                raise InventoryRefusal(
                    f"reparse point in protected ancestor chain: {current}"
                )

        parent = os.path.dirname(current)
        if parent == current:
            break
        current = parent
        first = False


def load_manifest(task_root: str) -> list[ManifestEntry]:
    manifest_path = os.path.join(task_root, "manifest.json")
    try:
        with open(extended_path(manifest_path), "r", encoding="utf-8-sig") as handle:
            document = json.load(handle)
    except (OSError, json.JSONDecodeError) as exc:
        raise InventoryRefusal(f"cannot load manifest.json: {exc}") from exc

    raw_entries = document.get("entries")
    if not isinstance(raw_entries, list):
        raise InventoryRefusal("manifest.json entries must be a list")

    result: list[ManifestEntry] = []
    for index, item in enumerate(raw_entries):
        if not isinstance(item, dict):
            raise InventoryRefusal(f"manifest entry {index} is not an object")

        source = item.get("source")
        destination = item.get("destination")
        if not isinstance(source, str) or not source:
            raise InventoryRefusal(f"manifest entry {index} has no valid source")

        if destination is None:
            continue
        if not isinstance(destination, str) or not destination:
            raise InventoryRefusal(
                f"manifest entry {index} has no usable proposed destination"
            )

        absolute_source = os.path.abspath(source)
        result.append(
            ManifestEntry(
                source=absolute_source,
                destination=os.path.abspath(destination),
                alias=os.path.basename(os.path.normpath(absolute_source)),
                status=item.get("status") if isinstance(item.get("status"), str) else None,
            )
        )

    return result


def resolve_source(requested: str, entries: list[ManifestEntry]) -> ManifestEntry:
    requested = requested.strip()
    if not requested:
        raise InventoryRefusal("--source must not be empty")

    has_path_syntax = (
        os.path.isabs(requested)
        or "\\" in requested
        or "/" in requested
        or ":" in requested
    )

    if has_path_syntax:
        matches = [entry for entry in entries if same_path(requested, entry.source)]
    else:
        matches = [
            entry for entry in entries
            if entry.alias.casefold() == requested.casefold()
        ]

    if not matches:
        raise InventoryRefusal(
            f"source {requested!r} is not an approved manifest source "
            "with a proposed destination"
        )
    if len(matches) != 1:
        raise InventoryRefusal(f"source alias {requested!r} is ambiguous")
    return matches[0]


def validate_source(entry: ManifestEntry, task_root: str) -> None:
    source = entry.source
    recovery_root = os.path.join(task_root, "recovery")

    if any(part in PROTECTED_NAMES for part in path_components(source)):
        raise InventoryRefusal("source is inside a protected credential-store path")
    if is_within(source, recovery_root):
        raise InventoryRefusal("task recovery trees may not be inventory sources")
    if not lexical_exists(source):
        raise InventoryRefusal(f"approved source does not exist: {source}")

    assert_no_reparse_ancestors(source, require_leaf=True)

    try:
        st = lstat(source)
    except OSError as exc:
        raise InventoryRefusal(f"cannot inspect approved source: {exc}") from exc

    if is_reparse(st):
        raise InventoryRefusal("approved source root is a reparse point")
    if not stat.S_ISDIR(st.st_mode):
        raise InventoryRefusal("approved source root is not a directory")

    \# A destination object of any kind is a collision; no merge is implied.
    if lexical_exists(entry.destination):
        raise InventoryRefusal(
            f"proposed destination already exists; collision refused: "
            f"{entry.destination}"
        )


def validate_prefix(prefix: str) -> str:
    if not PREFIX_RE.fullmatch(prefix) or prefix in {".", ".."}:
        raise InventoryRefusal(
            "--output-prefix must be a single safe filename stem containing "
            "only letters, digits, dot, underscore, or hyphen"
        )
    return prefix


def prepare_reports(task_root: str) -> str:
    reports = os.path.abspath(os.path.join(task_root, "reports"))
    if not is_within(reports, task_root):
        raise InventoryRefusal("reports directory escaped the task root")

    assert_no_reparse_ancestors(task_root, require_leaf=True)

    if lexical_exists(reports):
        assert_no_reparse_ancestors(reports, require_leaf=True)
        if not stat.S_ISDIR(lstat(reports).st_mode):
            raise InventoryRefusal("reports exists but is not a directory")
    else:
        os.mkdir(extended_path(reports))
        assert_no_reparse_ancestors(reports, require_leaf=True)

    return reports


def reserve_outputs(
    reports: str, prefix: str
) -> tuple[str, str, TextIO, TextIO]:
    records_path = os.path.join(reports, prefix + ".records.jsonl")
    summary_path = os.path.join(reports, prefix + ".summary.json")

    if not is_within(records_path, reports) or not is_within(summary_path, reports):
        raise InventoryRefusal("output path escaped reports")

    records_handle: TextIO | None = None
    try:
        records_handle = open(
            extended_path(records_path), "x", encoding="utf-8", newline="\n"
        )
        summary_handle = open(
            extended_path(summary_path), "x", encoding="utf-8", newline="\n"
        )
    except FileExistsError as exc:
        if records_handle is not None:
            records_handle.close()
            try:
                os.unlink(extended_path(records_path))
            except OSError:
                pass
        raise InventoryRefusal(
            f"report collision refused for output prefix {prefix!r}"
        ) from exc
    except OSError:
        if records_handle is not None:
            records_handle.close()
            try:
                os.unlink(extended_path(records_path))
            except OSError:
                pass
        raise

    return records_path, summary_path, records_handle, summary_handle


def emit(handle: TextIO, record: dict[str, Any]) -> None:
    handle.write(json.dumps(record, ensure_ascii=False, separators=(",", ":")))
    handle.write("\n")


def classify(st: os.stat_result, reparse: bool) -> tuple[str, str]:
    if stat.S_ISDIR(st.st_mode):
        base_kind = "directory"
    elif stat.S_ISREG(st.st_mode):
        base_kind = "file"
    elif stat.S_ISLNK(st.st_mode):
        base_kind = "symlink"
    else:
        base_kind = "other"
    return ("reparse" if reparse else base_kind), base_kind


def protected_descendant_reason(path: str, task_root: str) -> str | None:
    name = os.path.basename(os.path.normpath(path)).casefold()
    if name in PROTECTED_NAMES:
        return f"protected credential-store root {name}"

    recovery_root = os.path.join(task_root, "recovery")
    if same_path(path, recovery_root) or is_within(path, recovery_root):
        return "task recovery tree"

    return None


def scan(
    entry: ManifestEntry,
    task_root: str,
    records_handle: TextIO,
    max_entries: int,
    max_seconds: float,
) -> dict[str, Any]:
    started_wall = datetime.now(timezone.utc)
    started_clock = time.monotonic()
    deadline = started_clock + max_seconds
    state = ScanState()
    kind_counts: Counter[str] = Counter()
    stack = [entry.source]

    def error_record(path: str, relative_path: str | None, operation: str, exc: OSError) -> None:
        state.error_count += 1
        emit(
            records_handle,
            {
                "record_type": "error",
                "path": path,
                "relative_path": relative_path,
                "operation": operation,
                **safe_error(exc),
            },
        )

    while stack:
        if state.entry_count >= max_entries:
            state.stop_reason = "entry_limit"
            break
        if time.monotonic() >= deadline:
            state.stop_reason = "elapsed_limit"
            break

        path = stack.pop()
        try:
            st = lstat(path)
        except OSError as exc:
            relative = (
                "."
                if same_path(path, entry.source)
                else os.path.relpath(path, entry.source)
            )
            error_record(path, relative, "lstat", exc)
            continue

        relative = (
            "."
            if same_path(path, entry.source)
            else os.path.relpath(path, entry.source)
        )
        projected = (
            entry.destination
            if relative == "."
            else os.path.join(entry.destination, relative)
        )

        attrs = attributes_for(st)
        reparse = bool(attrs & REPARSE_ATTRIBUTE)
        kind, base_kind = classify(st, reparse)
        source_units = utf16_units(path)
        projected_units = utf16_units(projected)
        policy_reason = (
            None if relative == "." else protected_descendant_reason(path, task_root)
        )

        record: dict[str, Any] = {
            "record_type": "entry",
            "relative_path": relative,
            "source_path": path,
            "projected_path": projected,
            "kind": kind,
            "base_kind": base_kind,
            "size_bytes": int(st.st_size),
            "logical_file_bytes": (
                int(st.st_size) if base_kind == "file" and not reparse else 0
            ),
            "reparse_bytes": int(st.st_size) if reparse else 0,
            "attributes": attrs,
            "attribute_names": attribute_names(attrs),
            "is_reparse": reparse,
            "reparse_tag": getattr(st, "st_reparse_tag", None),
            "hardlink_count": int(st.st_nlink),
            "device_id": int(st.st_dev),
            "file_id": int(st.st_ino),
            "mode": int(st.st_mode),
            "mtime_ns": int(st.st_mtime_ns),
            "ctime_ns": int(st.st_ctime_ns),
            "atime_ns": int(st.st_atime_ns),
            "birthtime_ns": getattr(st, "st_birthtime_ns", None),
            "source_utf16_units": source_units,
            "projected_utf16_units": projected_units,
            "policy_skip_reason": policy_reason,
        }

        if reparse:
            try:
                record["reparse_target"] = os.readlink(extended_path(path))
            except OSError as exc:
                record["reparse_target"] = None
                error_record(path, relative, "readlink", exc)

        emit(records_handle, record)
        state.entry_count += 1
        kind_counts[kind] += 1
        state.max_source_utf16_units = max(
            state.max_source_utf16_units, source_units
        )
        state.max_projected_utf16_units = max(
            state.max_projected_utf16_units, projected_units
        )

        if reparse:
            state.reparse_count += 1
            state.reparse_bytes += int(st.st_size)
        elif base_kind == "file":
            state.file_bytes += int(st.st_size)

        if policy_reason is not None:
            state.policy_skip_count += 1
            continue
        if reparse or base_kind != "directory":
            continue

        \# Narrow the type-swap window before enumeration. A race remains after
        \# this check and is reported in the summary.
        try:
            before_scan = lstat(path)
        except OSError as exc:
            error_record(path, relative, "pre_scandir_lstat", exc)
            continue

        if (
            is_reparse(before_scan)
            or before_scan.st_dev != st.st_dev
            or before_scan.st_ino != st.st_ino
        ):
            state.error_count += 1
            emit(
                records_handle,
                {
                    "record_type": "error",
                    "path": path,
                    "relative_path": relative,
                    "operation": "pre_scandir_identity_check",
                    "winerror": None,
                    "errno": None,
                    "message": "directory identity/type changed during scan",
                },
            )
            continue

        children: list[str] = []
        remaining_capacity = max_entries - state.entry_count - len(stack)

        try:
            with os.scandir(extended_path(path)) as iterator:
                for item in iterator:
                    if time.monotonic() >= deadline:
                        state.stop_reason = "elapsed_limit"
                        state.enumeration_cutoff = True
                        break

                    if len(children) >= max(remaining_capacity, 0):
                        state.enumeration_cutoff = True
                        if state.stop_reason is None:
                            state.stop_reason = "entry_limit"
                        break

                    children.append(os.path.join(path, item.name))
        except OSError as exc:
            error_record(path, relative, "scandir", exc)
            continue

        \# Reverse so filesystem enumeration order is retained under a LIFO walk.
        stack.extend(reversed(children))

        if state.stop_reason == "elapsed_limit":
            break

    traversal_exhausted = (
        not stack
        and not state.enumeration_cutoff
        and state.stop_reason is None
    )
    coverage_complete = (
        traversal_exhausted
        and state.error_count == 0
        and state.policy_skip_count == 0
    )

    elapsed = time.monotonic() - started_clock
    finished_wall = datetime.now(timezone.utc)

    return {
        "schema_version": 1,
        "scanner": "scripts/inventory.py",
        "source_alias": entry.alias,
        "source": entry.source,
        "proposed_destination": entry.destination,
        "manifest_status": entry.status,
        "started_utc": started_wall.isoformat(),
        "finished_utc": finished_wall.isoformat(),
        "elapsed_seconds": elapsed,
        "limits": {
            "max_entries": max_entries,
            "max_seconds": max_seconds,
        },
        "entry_count": state.entry_count,
        "error_count": state.error_count,
        "policy_skip_count": state.policy_skip_count,
        "reparse_count": state.reparse_count,
        "kind_counts": dict(sorted(kind_counts.items())),
        "logical_file_bytes": state.file_bytes,
        "reparse_bytes": state.reparse_bytes,
        "max_source_utf16_units": state.max_source_utf16_units,
        "max_projected_utf16_units": state.max_projected_utf16_units,
        "traversal_exhausted": traversal_exhausted,
        "coverage_complete": coverage_complete,
        "stop_reason": state.stop_reason,
        "directory_enumeration_cutoff": state.enumeration_cutoff,
        "content_hashes_computed": False,
        "regular_file_contents_opened": False,
        "reparse_points_followed": False,
        "toctou_warning": TOCTOU_WARNING,
    }


def parser() -> argparse.ArgumentParser:
    result = argparse.ArgumentParser(
        description="Inventory one manifest-approved source without reading file contents."
    )
    result.add_argument(
        "--source",
        required=True,
        help="Exact manifest source path or unique manifest basename, e.g. atlas-design.",
    )
    result.add_argument(
        "--output-prefix",
        help="Safe filename prefix under reports; existing outputs are never overwritten.",
    )
    result.add_argument("--max-entries", type=int, default=1_000_000)
    result.add_argument("--max-seconds", type=float, default=3600.0)
    return result


def run(argv: list[str] | None = None, task_root: str | None = None) -> int:
    args = parser().parse_args(argv)
    if args.max_entries <= 0:
        raise InventoryRefusal("--max-entries must be positive")
    if args.max_seconds <= 0:
        raise InventoryRefusal("--max-seconds must be positive")

    if task_root is None:
        task_root = os.path.abspath(
            os.path.join(os.path.dirname(__file__), os.pardir)
        )
    else:
        \# Test-only dependency injection; the CLI exposes no task-root option.
        task_root = os.path.abspath(task_root)

    entries = load_manifest(task_root)
    entry = resolve_source(args.source, entries)
    validate_source(entry, task_root)

    prefix = args.output_prefix
    if prefix is None:
        stamp = datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%S.%fZ")
        prefix = f"{entry.alias}-{stamp}-{os.getpid()}"
    prefix = validate_prefix(prefix)

    reports = prepare_reports(task_root)
    (
        records_path,
        summary_path,
        records_handle,
        summary_handle,
    ) = reserve_outputs(reports, prefix)

    try:
        summary = scan(
            entry=entry,
            task_root=task_root,
            records_handle=records_handle,
            max_entries=args.max_entries,
            max_seconds=args.max_seconds,
        )
        records_handle.flush()
        summary["records_report"] = records_path
        summary["summary_report"] = summary_path
        json.dump(summary, summary_handle, ensure_ascii=False, indent=2)
        summary_handle.write("\n")
        summary_handle.flush()
    finally:
        records_handle.close()
        summary_handle.close()

    print(json.dumps({
        "records_report": records_path,
        "summary_report": summary_path,
        "traversal_exhausted": summary["traversal_exhausted"],
        "coverage_complete": summary["coverage_complete"],
    }))

    return 0 if summary["coverage_complete"] else 2


def main(argv: list[str] | None = None, task_root: str | None = None) -> int:
    try:
        return run(argv=argv, task_root=task_root)

def set_hidden(path: str) -> bool:
    if os.name != "nt":
        return False

    get_attributes = ctypes.windll.kernel32.GetFileAttributesW
    set_attributes = ctypes.windll.kernel32.SetFileAttributesW
    get_attributes.argtypes = [ctypes.c_wchar_p]
    get_attributes.restype = ctypes.c_uint32
    set_attributes.argtypes = [ctypes.c_wchar_p, ctypes.c_uint32]
    set_attributes.restype = ctypes.c_int

    if attrs == 0xFFFFFFFF:
        return False
    return bool(set_attributes(extended, attrs | 0x2))
