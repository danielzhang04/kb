```python
from __future__ import annotations
import argparse
from dataclasses import dataclass
from datetime import datetime, timedelta, timezone
import hashlib
import json
import secrets
import sqlite3
import uuid
from pathlib import Path

import cards
import scripts.stage_approval as stage_approval
from scripts.prospecting.approval.scope import BatchItem, build_scope, scope_hash, serialize
from scripts.prospecting.approval.verify import verify_and_insert

@dataclass(frozen=True)
class BatchResult:
    state: str
    ids: tuple[str,...]

def _identical_completed_batch(connection,expected_hash,assertion_ref):
    assertion_hash=hashlib.sha256(assertion_ref.encode()).hexdigest()
    receipt=connection.execute(
        "SELECT after_hash FROM audit WHERE action='p6_batch_materialized' "
        "AND entity_type='approval_batch' AND entity_id=? AND before_hash=?",
        (expected_hash,assertion_hash),
    ).fetchone()
    if receipt is None:
        return None
    rows=connection.execute(
        "SELECT a.approval_id,d.state,e.status FROM approval a "
        "JOIN delivery d ON d.campaign_id=a.campaign_id AND d.revision_hash=a.revision_hash "
        "AND d.contact_id=a.contact_id AND d.mailbox_id=a.mailbox_id "
        "JOIN enrollment e ON e.enrollment_id=d.enrollment_id "
        "WHERE a.assertion_ref=? ORDER BY a.approval_id",(assertion_ref,),
    ).fetchall()
    ids=tuple(row[0] for row in rows)
    digest=hashlib.sha256("\n".join(ids).encode()).hexdigest()
    if not ids or digest != receipt[0] or any(
        row[1] not in {"reserved","claimed","attempted","sent","uncertain"}
        or row[2] not in {"scheduled","sent","closed"} for row in rows
    ):
        raise ValueError("partial_or_nonidentical_replay")
    return ids

def assemble_due_batch(connection, campaign_id, mailbox_id, now, nonce):
    campaign = connection.execute(
        "SELECT policy_hash,send_window,timezone,approval_tier FROM campaign WHERE campaign_id=? "
        "AND status='active'", (campaign_id,)
    ).fetchone()
    if campaign is None or campaign[3] != "T1":
        raise ValueError("t1_campaign_required")
    candidates = connection.execute(
        "SELECT e.enrollment_id,r.hash,cp.contact_id FROM enrollment e "
        "JOIN revision r ON r.campaign_id=e.campaign_id AND r.person_id=e.person_id AND r.step=0 "
        "JOIN contact_point cp ON cp.person_id=e.person_id AND cp.state='valid' AND cp.selected=1 "
        "WHERE e.campaign_id=? AND e.current_step=0 AND e.status='approved' "
        "AND e.next_due_at<=? AND json_extract(r.qa,'$.qa_score')>=80 "
        "ORDER BY e.enrollment_id,r.hash,cp.contact_id", (campaign_id,now.isoformat())
    ).fetchall()
    grouped = {}
    for enrollment_id,revision_hash,contact_id in candidates:
        grouped.setdefault(enrollment_id,[]).append((revision_hash,contact_id))
    for enrollment_id,selected in grouped.items():
        if len(selected) != 1:
            raise ValueError(f"selected_contact_count:{enrollment_id}")
    rows = [selected[0] for selected in grouped.values()]
    if not rows:
        raise ValueError("empty_batch")
    start, end = _next_business_window(now, campaign[1], campaign[2])
    if end > now + timedelta(hours=24):
        raise ValueError("scope_prepared_too_early")
    expiry = end
    return build_scope(
        campaign_id,campaign[0],tuple(BatchItem(*row) for row in rows),mailbox_id,"T1",
        start.isoformat(),end.isoformat(),expiry.isoformat(),nonce,now,
    )

def _next_business_window(now, window, timezone_name):
    start_text,end_text = window.split("-",1)
    zone = ZoneInfo(timezone_name)
    day = (now.astimezone(zone) + timedelta(days=1)).date()
    while day.weekday() >= 5:
        day += timedelta(days=1)
    start = datetime.combine(day,datetime.strptime(start_text,"%H:%M").time(),zone)
    end = datetime.combine(day,datetime.strptime(end_text,"%H:%M").time(),zone)
    return start.astimezone(timezone.utc),end.astimezone(timezone.utc)

def summarize(scope):
    return {
        "campaign_id":scope.campaign_id,"batch_hash":scope.batch_hash,
        "scope_hash":scope_hash(scope),"count":len(scope.items),
        "revision_ids":[item.revision_hash for item in scope.items],
        "contact_ids":[item.contact_id for item in scope.items],
    }

def stage_scope(scope, repo_root, opener):
    card_id=f"prospecting-t1-{scope.batch_hash[:16]}"
    text=(
        "---\n"
        f"id: {card_id}\n"
        "action: approve-prospecting-send\n"
        f"target: {scope.campaign_id}\n"
        "risk-tier: T3\n"
        "owner: daniel\n"
        "---\n\n"
        "## Work order\n\n"
        f"{serialize(scope)}\n"
    )
    card=cards.parse(text)
    return stage_approval.stage(card,repo_root=repo_root,opener=opener)

def apply_batch(connection, assertion_ref, card_path, credentials_path, expected_hash, now):
    try:
        connection.execute("BEGIN IMMEDIATE")
        try:
            ids = verify_and_insert(
                connection,assertion_ref,card_path,credentials_path,expected_hash,now,
                manage_transaction=False,
            )
        except ValueError as exc:
            if str(exc) != "approval_replay":
                raise
            completed=_identical_completed_batch(connection,expected_hash,assertion_ref)
            if completed is None:
                raise ValueError("partial_or_nonidentical_replay") from exc
            connection.rollback()
            return BatchResult("identical_completed_noop",completed)
        for approval_id in ids:
            row = connection.execute(
                "SELECT campaign_id,revision_hash,contact_id,mailbox_id,send_window "
                "FROM approval WHERE approval_id=?", (approval_id,)
            ).fetchone()
            campaign_id,revision_hash,contact_id,mailbox_id,window_json = row
            enrollment_id = connection.execute(
                "SELECT e.enrollment_id FROM enrollment e JOIN revision r "
                "ON r.person_id=e.person_id AND r.campaign_id=e.campaign_id "
                "WHERE e.campaign_id=? AND r.hash=? AND e.current_step=0 "
                "AND e.status='approved' AND e.stop_reason IS NULL AND e.block_reason IS NULL "
                "AND json_extract(r.qa,'$.qa_score')>=80",
                (campaign_id,revision_hash),
            ).fetchone()[0]
            logical = hashlib.sha256(
                f"{campaign_id}:{enrollment_id}:0:{revision_hash}:{contact_id}:{mailbox_id}".encode()
            ).hexdigest()
            start = json.loads(window_json)["start"]
            connection.execute(
                "INSERT INTO delivery(delivery_id,campaign_id,enrollment_id,step,revision_hash,"
                "contact_id,mailbox_id,logical_key,rfc_message_id,scheduled_at,state) "
                "VALUES(?,?,?,?,?,?,?,?,?,?,?) ON CONFLICT(logical_key) DO NOTHING",
                (logical,campaign_id,enrollment_id,0,revision_hash,contact_id,mailbox_id,
                 logical,f"<{logical}@prospecting.test>",start,"reserved"),
            )
            connection.execute(
                "UPDATE enrollment SET status='scheduled',block_reason=NULL "
                "WHERE enrollment_id=? AND status='approved'", (enrollment_id,)
            )
        assertion_hash=hashlib.sha256(assertion_ref.encode()).hexdigest()
        ids_hash=hashlib.sha256("\n".join(sorted(ids)).encode()).hexdigest()
        connection.execute(
            "INSERT INTO audit(event_id,actor,action,entity_type,entity_id,at,before_hash,after_hash,reason) "
            "VALUES(?,?,?,?,?,?,?,?,?)",
            (str(uuid.uuid4()),"prospecting-desktop","p6_batch_materialized","approval_batch",
             expected_hash,now.isoformat(),assertion_hash,ids_hash,"verified_complete"),
        )
        connection.commit()
        return BatchResult("materialized",tuple(ids))
    except Exception:
        connection.rollback()
        raise

def main(argv=None, connection=None, opener=None):
    parser=argparse.ArgumentParser()
    sub=parser.add_subparsers(dest="command",required=True)
    prepare=sub.add_parser("prepare")
    prepare.add_argument("--campaign",required=True); prepare.add_argument("--mailbox",required=True)
    prepare.add_argument("--repo-root",type=Path,required=True)
    apply=sub.add_parser("apply")
    for name in ("assertion-file","card","credentials","scope-hash"):
        apply.add_argument(f"--{name}",required=True)
    args=parser.parse_args(argv)
    if connection is None:
        from scripts.prospecting.store import open_store
        connection = open_store()
    now=datetime.now(timezone.utc)
    if args.command=="prepare":
        scope=assemble_due_batch(connection,args.campaign,args.mailbox,now,secrets.token_urlsafe(32))
        if opener is None:
            raise RuntimeError("approval_pr_opener_required")
        staged=stage_scope(scope,args.repo_root,opener)
        print(json.dumps({"staged_ref":staged.ref,"summary":summarize(scope)},sort_keys=True))
    else:
        assertion_ref=Path(args.assertion_file).read_text(encoding="utf-8").strip()
        if not assertion_ref or "\n" in assertion_ref:
            raise ValueError("invalid_assertion_reference")
        ids=apply_batch(
            connection,assertion_ref,Path(args.card),Path(args.credentials),
            args.scope_hash,now
        )
        print(json.dumps({"state":ids.state,"approved":len(ids.ids),"approval_ids":list(ids.ids)},sort_keys=True))
    return 0
```
