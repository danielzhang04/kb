        return result

    if git_entry["kind"] == "file":
        result["gitForm"] = "file"
    elif git_entry["kind"] == "dir":
        result["gitForm"] = "directory"
    else:
        result["gitForm"] = "unsupported"
        result["unknowns"].append("git-form-unsupported")
        return result

    git_prefix = git_rel.casefold()
    hooks_prefix = git_prefix + "\\hooks\\"
    hooks_dir = entries.get((git_rel + "\\hooks").casefold())
    result["hooksPresent"] = bool(
        hooks_dir is not None and hooks_dir["kind"] == "dir"
    )
    for folded, entry in entries.items():
        if not folded.startswith(hooks_prefix):
            continue
        tail = entry["path"][len(git_rel) + len("\\hooks\\"):]
        if "\\" in tail or entry["kind"] != "file":
            continue
        if tail.casefold().endswith(".sample"):
            result["sampleHookNameCount"] += 1
        else:
            result["activeHookNameCount"] += 1

    pointer_map = {item["role"]: item for item in result["pointers"]}
    selected: list[tuple[str, dict[str, Any]]] = []

    if result["gitForm"] == "file":
        selected.append(("gitdir", git_entry))
        result["unknowns"].append("config-not-located-through-gitfile")
    else:
        config_rel = git_rel + "\\config"
        worktree_config_rel = git_rel + "\\config.worktree"
        for label, rel in (
            ("config", config_rel),
            ("configWorktree", worktree_config_rel),
        ):
            entry = entries.get(rel.casefold())
            if entry is None:
                result[label] = empty_config("absent")
                result["unknowns"].append(label + "-absent")
            elif entry["kind"] != "file":
                result[label] = empty_config("unsupported")
                result["unknowns"].append(label + "-unsupported")
            else:
                selected.append((label, entry))

        for folded, entry in entries.items():
            role = pointer_role(folded, git_prefix)
            if role is not None:
                if entry["kind"] != "file":
                    result["unknowns"].append("pointer-kind-unsupported")
                else:
                    selected.append((role, entry))

    selected.sort(key=lambda item: (item[0], item[1]["path"].casefold()))
    for role, entry in selected:
        check_deadline(deadline)
        size = entry["length"]
        if size > FILE_LIMIT:
            fail("selected-file-size-limit")
        budget.add(size)
        path = entry_disk_path(recovery_root, entry["path"])
        data, _ = read_guarded(
            path,
            FILE_LIMIT,
            deadline,
            expected_size=size,
            expected_sha256=entry["sha256"],
        )

        if role in {"config", "configWorktree"}:
            result[role] = scan_config(data)
            continue

        base_parts = tuple(entry["path"].split("\\")[:-1])
        counts, complex_syntax = classify_pointer(role, data, base_parts)
        summary = pointer_map[role]
        summary["fileCount"] += 1
        for classification in POINTER_CLASSES:
            summary[classification] += counts[classification]
        summary["complexSyntax"] = (
            summary["complexSyntax"] or complex_syntax
        )

    review = (
        result["gitForm"] != "directory"
        or result["activeHookNameCount"] > 0
        or bool(result["unknowns"])
    )
    for label in ("config", "configWorktree"):
        config = result[label]
        review = review or config["state"] == "unsupported"
        review = review or config["complexSyntax"]
        review = review or any(
            config[key]
            for key in (
                "include",
                "commandBearing",
                "externalStore",
                "credential",
                "pathKey",
            )
        )
    for pointer in result["pointers"]:
        review = review or pointer["complexSyntax"]
        review = review or any(
            pointer[classification] > 0
            for classification in (
                "relative-escape",
                "absolute",
                "network",
                "unknown",
            )
        )
    result["requiresLocalReview"] = review
    return result


def report_template(manifest_sha256: str) -> dict[str, Any]:
    return {
        "schema": "ait25-recovery-config-inspection/1",
        "status": "FAIL",
        "assessment": "observation-only",
        "gitEligibility": "not-assessed",
        "manifestSha256": manifest_sha256,
        "pathSafety": (
            "ancestor-lstat-and-pre-post-identity-pathname-check;"
            "not-atomic-confinement"
        ),
        "limits": {
            "deadlineSeconds": DEADLINE_SECONDS,
            "manifestBytes": MANIFEST_LIMIT,
            "selectedFileBytes": FILE_LIMIT,
            "selectedFileCount": FILE_COUNT_LIMIT,
            "selectedTotalBytes": TOTAL_READ_LIMIT,
        },
        "selectedFileCount": 0,
        "selectedTotalBytes": 0,
        "requiresLocalReview": True,
        "unknowns": [],
        "repositories": [
            default_repository(name) for name in REPOSITORIES
        ],
        "failureCode": None,
    }


def create_report(report: dict[str, Any]) -> Path:
    check_directory_chain(REPORTS)
    path = REPORTS / (
        "ait25-config-inspection-" + uuid.uuid4().hex + ".json"
    )
    payload = (
        json.dumps(
            report,
            ensure_ascii=True,
            indent=2,
            sort_keys=True,
            allow_nan=False,
        )
        + "\n"
    ).encode("utf-8")
    try:
        with path.open("xb", buffering=0) as stream:
            stream.write(payload)
            stream.flush()
            os.fsync(stream.fileno())
    except OSError:
        fail("report-create-failed")
    return path


def parse_cli(argv: list[str]) -> tuple[str, str] | str:
    if argv == ["--self-test"]:
        return "self-test"
    if len(argv) != 4:
        fail("invalid-cli")
    values: dict[str, str] = {}
    index = 0
    while index < len(argv):
        option = argv[index]
        if option not in {"--generation", "--manifest-sha256"}:
            fail("invalid-cli")
        if option in values:
            fail("invalid-cli")
        values[option] = argv[index + 1]
        index += 2
    if set(values) != {"--generation", "--manifest-sha256"}:
        fail("invalid-cli")
    return values["--generation"], values["--manifest-sha256"]


def self_test() -> int:
    def refused(rel: str) -> bool:
        try:
            validate_relative_path(rel)
            return False
        except InspectionFailure:
            return True

    if not all(
        (
            refused(r"..\escape"),
            refused(r"folder\..\escape"),
            refused(r"file:stream"),
            refused(r"C:\absolute"),
        )
    ):
        return 2

    config_sentinel = "SENTINEL_CONFIG_SECRET"
    pointer_sentinel = "SENTINEL_POINTER_SECRET"
    config = scan_config(
        (
            '[include]\npath = "C:/'
            + config_sentinel
            + '"\n[credential]\nhelper = !'
            + config_sentinel
            + "\n[core]\nsshCommand = "
            + config_sentinel
            + "\n"
        ).encode("utf-8")
    )
    if not (
        config["include"]
        and config["commandBearing"]
        and config["credential"]
        and config["pathKey"]
    ):
        return 2

    cases = {
        r"inside\objects": "relative-internal",
        r"..\..\..\escape": "relative-escape",
        r"C:\outside": "absolute",
        r"\\server\share": "network",
        r"https://example.invalid/store": "network",
        r"value:stream": "unknown",
    }
    for value, expected in cases.items():
        if lexical_pointer_class(value, ("repo", ".git")) != expected:
            return 2

    redacted = {
        "config": config,
        "pointer": {
            "classification": lexical_pointer_class(
                pointer_sentinel, ("repo", ".git")
            )
        },
    }
    serialized = json.dumps(redacted, sort_keys=True)
    if config_sentinel in serialized or pointer_sentinel in serialized:
        return 2

    print(json.dumps({"status": "SELF-TEST-PASS", "reportPath": None}))
    return 0


def inspect(generation_text: str, manifest_sha256: str) -> int:
    report = report_template(manifest_sha256)
    report_path: Path | None = None
    deadline = time.monotonic() + DEADLINE_SECONDS

    try:
        if not UPPER_SHA.fullmatch(manifest_sha256):
            fail("invalid-manifest-sha256")
        if not re.fullmatch(
            re.escape(str(GENERATIONS)) + r"\\g-[0-9a-f]{32}",
            generation_text,
        ):
            fail("invalid-generation")
        generation = Path(generation_text)
        if str(generation) != generation_text:
            fail("invalid-generation")
        validate_path_text(generation)
        check_directory_chain(generation)

        recovery = generation / "recovery"
        recovery_root = recovery / "parent" / "ait25"
        check_directory_chain(recovery)
        check_directory_chain(recovery_root)

        manifest_path = recovery / "manifest.json"
        manifest_data, manifest_before = read_guarded(
            manifest_path,
            MANIFEST_LIMIT,
            deadline,
            expected_sha256=manifest_sha256,
        )
        manifest = strict_json(manifest_data)
        entries, _ = validate_manifest(manifest, generation)

        budget = Budget()
        repositories = [
            inspect_repository(
                name,
                entries,
                recovery_root,
                budget,
                deadline,
            )
            for name in REPOSITORIES
        ]

        check_deadline(deadline)
        _, manifest_after = read_guarded(
            manifest_path,
            MANIFEST_LIMIT,
            deadline,
            expected_size=manifest_before.st_size,
            expected_sha256=manifest_sha256,
        )
        if not same_observation(manifest_before, manifest_after):
            fail("manifest-postread-changed")

        report["repositories"] = repositories
        report["selectedFileCount"] = budget.files
        report["selectedTotalBytes"] = budget.bytes
        report["requiresLocalReview"] = any(
            item["requiresLocalReview"] for item in repositories
        )
        report["unknowns"] = sorted(
            {
                unknown
                for item in repositories
                for unknown in item["unknowns"]
            }
        )
        report["status"] = "COMPLETE"
        report["failureCode"] = None
        report_path = create_report(report)
        print(
            json.dumps(
                {"status": "COMPLETE", "reportPath": str(report_path)}
            )
        )
        return 0
    except InspectionFailure as exc:
        report["status"] = "FAIL"
        report["failureCode"] = exc.code
    except Exception:
        report["status"] = "FAIL"
        report["failureCode"] = "internal-error"

    try:
        report_path = create_report(report)
    except Exception:
        print(json.dumps({"status": "FAIL", "reportPath": None}))
        return 2
    print(json.dumps({"status": "FAIL", "reportPath": str(report_path)}))
    return 2


def main() -> int:
    try:
        parsed = parse_cli(sys.argv[1:])
    except InspectionFailure:
        print(json.dumps({"status": "FAIL", "reportPath": None}))
        return 2

    if parsed == "self-test":
        return self_test()

    if (
        os.name != "nt"
        or sys.version_info[:2] != (3, 12)
        or bool(ctypes.windll.shell32.IsUserAnAdmin())
    ):
        print(json.dumps({"status": "FAIL", "reportPath": None}))
        return 2

    generation, manifest_sha256 = parsed
    return inspect(generation, manifest_sha256)


if __name__ == "__main__":
    raise SystemExit(main())
