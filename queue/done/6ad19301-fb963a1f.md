def terminate_owned_tree(proc: subprocess.Popen[bytes]) -> dict[str, Any]:
    no_reparse_existing_chain(TASKKILL_EXE)
    lstat_regular(TASKKILL_EXE, "taskkill executable")
    try:
        result = subprocess.run(
            [str(TASKKILL_EXE), "/PID", str(proc.pid), "/T", "/F"],
            stdin=subprocess.DEVNULL,
            stdout=subprocess.DEVNULL,
            stderr=subprocess.DEVNULL,
            shell=False,
            timeout=5,
            check=False,
        )
        taskkill_code: int | None = result.returncode
        taskkill_error: str | None = None
    except (OSError, subprocess.TimeoutExpired) as exc:
        taskkill_code = None
        taskkill_error = type(exc).__name__
    try:
        proc.wait(timeout=5)
    except subprocess.TimeoutExpired:
        pass
    return {
        "taskkillReturnCode": taskkill_code,
        "taskkillErrorClass": taskkill_error,
        "treeExitObserved": proc.poll() is not None,
    }


def run_bounded(
    argv: list[str],
    cwd: Path,
    env: dict[str, str],
    logs: Path,
    expected: set[int] | frozenset[int] | None = frozenset({0}),
) -> tuple[int, bytes, bytes]:
    global command_counter
    command_counter += 1
    stem = f"{command_counter:03d}"
    stdout_path = logs / f"{stem}.stdout"
    stderr_path = logs / f"{stem}.stderr"

    out = stdout_path.open("xb")
    err = stderr_path.open("xb")
    proc: subprocess.Popen[bytes] | None = None
    timed_out = False
    overflow = False
    termination: dict[str, Any] | None = None
    code: int | None = None
    try:
        proc = subprocess.Popen(
            argv,
            cwd=cwd,
            env=env,
            stdin=subprocess.DEVNULL,
            stdout=out,
            stderr=err,
            shell=False,
        )
        deadline = time.monotonic() + PROCESS_TIMEOUT
        while True:
            if (
                os.fstat(out.fileno()).st_size > MAX_OUTPUT
                or os.fstat(err.fileno()).st_size > MAX_OUTPUT
            ):
                overflow = True
                termination = terminate_owned_tree(proc)
                break
            code = proc.poll()
            if code is not None:
                break
            if time.monotonic() >= deadline:
                timed_out = True
                termination = terminate_owned_tree(proc)
                break
            time.sleep(0.05)
        code = proc.poll()
    finally:
        out.close()
        err.close()

    stdout_size = lstat_regular(stdout_path, "stdout log").st_size
    stderr_size = lstat_regular(stderr_path, "stderr log").st_size
    watchdogs.append({
        "commandOrdinal": command_counter,
        "timeout": timed_out,
        "outputOverflow": overflow,
        "returnCode": code,
        "stdoutBytes": stdout_size,
        "stderrBytes": stderr_size,
        "termination": termination,
    })
    if timed_out:
        raise Refusal("child-timeout")
    if overflow:
        raise Refusal("child-output-overflow")
    if code is None:
        raise Refusal("child-termination-unobserved")
    stdout = read_regular_bounded(stdout_path, MAX_OUTPUT, "stdout log")[0]
    stderr = read_regular_bounded(stderr_path, MAX_OUTPUT, "stderr log")[0]
    if expected is not None and code not in expected:
        raise Refusal("child-exit", f"unexpected exit {code}")
    return code, stdout, stderr

def config_class(key: str) -> str:
    lowered = key.lower()
    if lowered in SAFE_CONFIG_KEYS or lowered in COMMAND_CONFIG_KEYS:
        return "fixed"
    if lowered.startswith("remote.") and lowered.endswith((".url", ".fetch")):
        return "remote"
    if lowered.startswith("branch.") and lowered.endswith((".remote", ".merge")):
        return "branch"
    raise Refusal("unexpected-config-key")


def parse_config(data: bytes) -> dict[str, Any]:
    fields = data.split(b"\0")
    if fields and fields[-1] == b"":
        fields.pop()
    if len(fields) % 3:
        raise Refusal("malformed-output", "config output field count")
    scopes: Counter[str] = Counter()
    classes: Counter[str] = Counter()
    for offset in range(0, len(fields), 3):
        scope = decode_utf8(fields[offset], "config scope")
        decode_utf8(fields[offset + 1], "config origin")
        key = decode_utf8(fields[offset + 2], "config name")
        if scope not in {"local", "command"}:
            raise Refusal("unexpected-config-scope")
        scopes[scope] += 1
        classes[config_class(key)] += 1
    return {
        "entryCount": sum(scopes.values()),
        "scopeCounts": {key: scopes[key] for key in ("local", "command")},
        "keyClassCounts": {
            key: classes[key] for key in ("fixed", "remote", "branch")
        },
    }


def parse_rev_parse(data: bytes, repository: Path) -> dict[str, Any]:
    lines = decode_utf8(data, "rev-parse").splitlines()
    if len(lines) != 5:
        raise Refusal("malformed-output", "rev-parse line count")
    if lines[0] not in {".git", str(repository / ".git")}:
        raise Refusal("repository-binding")
    if Path(os.path.normcase(os.path.abspath(lines[1]))) != Path(
        os.path.normcase(os.path.abspath(repository))
    ):
        raise Refusal("repository-binding")
    if lines[2] != "false" or lines[3] not in {"true", "false"}:
        raise Refusal("malformed-output", "rev-parse booleans")
    if lines[4] not in {"sha1", "sha256"}:
        raise Refusal("unsupported-object-format")
    return {
        "bare": False,
        "shallow": lines[3] == "true",
        "objectFormat": lines[4],
    }


def parse_head(
    symbolic_code: int,
    symbolic_output: bytes,
    commit_code: int,
    commit_output: bytes,
) -> str:
    symbolic = decode_utf8(symbolic_output, "symbolic HEAD").strip()
    commit = decode_utf8(commit_output, "commit verification").strip()
    if symbolic_code == 0:
        if not symbolic.startswith("refs/heads/") or "\n" in symbolic:
            raise Refusal("malformed-output", "symbolic HEAD")
        if commit_code == 0 and OID_RE.fullmatch(commit):
            return "symbolic-attached"
        if commit_code == 1 and not commit:
            return "symbolic-unborn"
    elif symbolic_code == 1 and not symbolic:
        if commit_code == 0 and OID_RE.fullmatch(commit):
            return "detached"
    raise Refusal("malformed-output", "inconsistent HEAD state")


def parse_refs(data: bytes) -> dict[str, Any]:
    namespaces = Counter()
    object_types = Counter()
    tracking = Counter()
    text = decode_utf8(data, "refs")
    if text and not text.endswith("\n"):
        raise Refusal("malformed-output", "unterminated ref row")
    for line in text.splitlines():
        fields = line.split("\t")
        if len(fields) != 4:
            raise Refusal("malformed-output", "ref field count")
        refname, object_type, upstream, track = fields
        if not refname.startswith("refs/"):
            raise Refusal("malformed-output", "ref namespace")
        if refname.startswith("refs/heads/"):
            namespace = "heads"
        elif refname.startswith("refs/remotes/"):
            namespace = "remotes"
        elif refname.startswith("refs/tags/"):
            namespace = "tags"
        elif refname == "refs/stash":
            namespace = "stash"
        elif refname.startswith("refs/replace/"):
            namespace = "replace"
        else:
            namespace = "other"
        namespaces[namespace] += 1
        object_types[object_type if object_type in {
            "commit", "tree", "blob", "tag"
        } else "other"] += 1

        if namespace == "heads":
            if not upstream:
                tracking["noUpstream"] += 1
            elif track == "gone":
                tracking["gone"] += 1
            elif track:
                matched = re.fullmatch(
                    r"(?:ahead ([0-9]+))?(?:, )?(?:behind ([0-9]+))?", track
                )
                if not matched or not any(matched.groups()):
                    raise Refusal("malformed-output", "tracking state")
                if matched.group(1) and int(matched.group(1)) > 0:
                    tracking["ahead"] += 1
                if matched.group(2) and int(matched.group(2)) > 0:
                    tracking["behind"] += 1
    return {
        "count": sum(namespaces.values()),
        "namespaceCounts": {
            key: namespaces[key]
            for key in ("heads", "remotes", "tags", "stash", "replace", "other")
        },
        "objectTypeCounts": {
            key: object_types[key]
            for key in ("commit", "tree", "blob", "tag", "other")
        },
        "branchTrackingCounts": {
            key: tracking[key]
            for key in ("ahead", "behind", "gone", "noUpstream")
        },
    }


def parse_stash(data: bytes) -> dict[str, int]:
    items = [item for item in data.split(b"\0") if item]
    for item in items:
        if not OID_RE.fullmatch(decode_utf8(item, "stash OID")):
            raise Refusal("malformed-output", "stash OID")
    return {"count": len(items)}


def parse_ls_files(
    data: bytes, lx_paths: set[str]
) -> dict[str, Any]:
    modes = Counter()
    entries = 0
    unmerged = 0
    tracked_lx = 0
    tracked_node_modules = 0
    for row in (item for item in data.split(b"\0") if item):
        try:
            header, raw_path = row.split(b"\t", 1)
            mode_raw, oid_raw, stage_raw = header.split(b" ")
        except ValueError as exc:
            raise Refusal("malformed-output", "ls-files row") from exc
        mode = decode_utf8(mode_raw, "index mode")
        oid = decode_utf8(oid_raw, "index OID")
        stage = decode_utf8(stage_raw, "index stage")
        path = decode_utf8(raw_path, "index path")
        if not re.fullmatch(r"[0-7]{6}", mode) or not OID_RE.fullmatch(oid):
            raise Refusal("malformed-output", "index entry")
        if stage not in {"0", "1", "2", "3"} or not path or "\x00" in path:
            raise Refusal("malformed-output", "index entry")
        entries += 1
        modes[mode if mode in {
            "100644", "100755", "120000", "160000"
        } else "other"] += 1
        unmerged += stage != "0"
        tracked_lx += path in lx_paths
        tracked_node_modules += "node_modules" in PurePosixPath(path).parts
    return {
        "entryCount": entries,
        "modeCounts": {
            key: modes[key]
            for key in ("100644", "100755", "120000", "160000", "other")
        },
        "unmergedStageCount": unmerged,
        "trackedLxCount": tracked_lx,
        "trackedNodeModulesCount": tracked_node_modules,
        "gitlinkWithoutGitmodules": modes["160000"] > 0,
    }


def parse_diff_index(data: bytes) -> dict[str, int]:
    fields = [field for field in data.split(b"\0") if field]
    if len(fields) % 2:
        raise Refusal("malformed-output", "diff-index field count")
    counts = Counter()
    for offset in range(0, len(fields), 2):
        status = decode_utf8(fields[offset], "diff status")
        decode_utf8(fields[offset + 1], "diff path")
        if status not in {"A", "M", "D", "T"}:
            raise Refusal("malformed-output", "unsupported diff status")
        counts[status] += 1
    return {key: counts[key] for key in ("A", "M", "D", "T")}

    "packs",

    "garbage",
    "size-garbage",

def parse_count_objects(data: bytes) -> dict[str, int]:
    values: dict[str, int] = {}
    for line in decode_utf8(data, "count-objects").splitlines():
        if ": " not in line:
            raise Refusal("malformed-output", "count-objects row")
        key, raw_value = line.split(": ", 1)
        if key not in COUNT_OBJECT_KEYS or key in values:
            raise Refusal("malformed-output", "count-objects key")
        if not raw_value.isdecimal():
            raise Refusal("malformed-output", "count-objects value")
        values[key] = int(raw_value)
    if set(values) != set(COUNT_OBJECT_KEYS):
        raise Refusal("malformed-output", "count-objects keys incomplete")
    return values


def parse_fsck(code: int, stdout: bytes, stderr: bytes) -> dict[str, Any]:
    text = decode_utf8(stdout + stderr, "fsck")
    counts = Counter()
    for line in text.splitlines():
        lowered = line.lower()
        if lowered.startswith("fatal:"):
            category = "fatal"
        elif lowered.startswith("error:"):
            category = "error"
        elif lowered.startswith("warning:"):
            category = "warning"
        elif lowered.startswith("notice:"):
            category = "notice"
        elif "missing" in lowered:
            category = "missing"
        elif "broken" in lowered:
            category = "broken"
        elif "dangling" in lowered:
            category = "dangling"
        else:
            category = "unclassified"
        counts[category] += 1
    finding = code != 0 or bool(text)
    return {
        "exitCode": code,
        "classification": "FINDING" if finding else "PASS",
        "messageClassCounts": {
            key: counts[key]
            for key in (
                "fatal", "error", "warning", "notice",
                "missing", "broken", "dangling", "unclassified",
            )
        },
    }


def ensure_time(started: float) -> None:
    if time.monotonic() - started >= COOPERATIVE_DEADLINE:
        raise Refusal("cooperative-deadline")


def inspect_repository(
    repository: str,
    copy_ait25: Path,
    run_root: Path,
    env: dict[str, str],
    logs: Path,
    entries: dict[str, dict[str, Any]],
    started: float,
) -> dict[str, Any]:
    repo = copy_ait25 / repository

    def run(
        *args: str,
        expected: set[int] | frozenset[int] | None = frozenset({0}),
    ) -> tuple[int, bytes, bytes]:
        ensure_time(started)
        return run_bounded(
            git_argv(run_root, repo, *args), repo, env, logs, expected
        )

    _, output, _ = run(
        "config", "--list", "--show-scope", "--show-origin", "--name-only", "-z"
    )
    config_result = parse_config(output)

    _, output, _ = run(
        "rev-parse",
        "--git-dir",
        "--show-toplevel",
        "--is-bare-repository",
        "--is-shallow-repository",
        "--show-object-format=storage",
    )
    repo_result = parse_rev_parse(output, repo)

    symbolic_code, symbolic_out, _ = run(
        "symbolic-ref", "-q", "HEAD", expected={0, 1}
    )
    commit_code, commit_out, _ = run(
        "rev-parse", "-q", "--verify", "HEAD^{commit}", expected={0, 1}
    )
    head_state = parse_head(
        symbolic_code, symbolic_out, commit_code, commit_out
    )

    _, output, _ = run(
        "for-each-ref",
        "--format=%(refname)%09%(objecttype)%09%(upstream)%09"
        "%(upstream:track,nobracket)",
    )
    refs_result = parse_refs(output)

    _, output, _ = run("stash", "list", "--format=%H", "-z")
    stash_result = parse_stash(output)

    _, output, _ = run("ls-files", "-z", "--stage")
    index_result = parse_ls_files(
        output, manifest_lx_paths(entries, repository)
    )

    if head_state == "symbolic-unborn":
        staged_result: dict[str, Any] = {"state": "skipped-unborn"}
    else:
        _, output, _ = run(
            "diff-index",
            "--cached",
            "-z",
            "--name-status",
            "--no-renames",
            "--no-ext-diff",
            "--no-textconv",
            "--ignore-submodules=all",
            "HEAD",
        )
        staged_result = {
            "state": "measured",
            "counts": parse_diff_index(output),
        }

    _, output, _ = run("count-objects", "-v")
    objects_result = parse_count_objects(output)

    fsck_code, fsck_out, fsck_err = run(
        "fsck",
        "--full",
        "--strict",
        "--no-progress",
        "--no-dangling",
        expected=None,
    )
    fsck_result = parse_fsck(fsck_code, fsck_out, fsck_err)

    return {
        "repository": repository,
        "config": config_result,
        "repositoryState": repo_result,
        "headState": head_state,
        "refs": refs_result,
        "stash": stash_result,
        "index": index_result,
        "staged": staged_result,
        "objects": objects_result,
        "fsck": fsck_result,
    }


def is_nonadmin_windows_user() -> bool:
    if os.name != "nt":
        return False
    try:
        import ctypes
        return not bool(ctypes.windll.shell32.IsUserAnAdmin())
    except Exception as exc:
        raise Refusal("privilege-check", type(exc).__name__) from exc


def pure_gate(
    copy_sha: str,
    manifest_sha: str,
    copier_sha: str,
    git_sha: str,
    config_matches: bool,
    runner: Callable[[], None],
) -> None:
    if copy_sha != "A" * 64:
        raise Refusal("copy-report-hash")
    if manifest_sha != MANIFEST_SHA:
        raise Refusal("manifest-hash")
    if copier_sha != COPIER_SHA:
        raise Refusal("copier-hash")
    if git_sha != GIT_SHA:
        raise Refusal("git-hash")
    if not config_matches:
        raise Refusal("config-binding")
    runner()
