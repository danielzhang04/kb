```python
from __future__ import annotations
from dataclasses import dataclass
import hashlib, json, sqlite3, uuid
from typing import Mapping, Sequence
from scripts.prospecting.store import FitScoreVersion, decide_eligibility, insert_fit_score_version

FIT_SCORE_V1_RULES={"title_match":30,"seniority_match":25,"industry_match":25,"location_match":20}

@dataclass(frozen=True)
class CleaningResult: valid:int; quarantined:int; excluded_role:int
@dataclass(frozen=True)
class DedupeResult: canonical:int; duplicates:int; conflicts:int

def clean_contacts(connection: sqlite3.Connection, person_ids: Sequence[str], at: str) -> CleaningResult:
    marks=",".join("?" for _ in person_ids)
    rows=connection.execute(f"SELECT state,COUNT(*) FROM contact_point WHERE person_id IN ({marks}) GROUP BY state",tuple(person_ids)).fetchall()
    counts=dict(rows); valid=int(counts.get("valid",0)); role=int(counts.get("role",0)); quarantined=sum(int(counts.get(s,0)) for s in ("invalid","risky","catch_all","role","stale"))
    return CleaningResult(valid,quarantined,role)

def dedupe_people(connection: sqlite3.Connection, person_ids: Sequence[str], at: str) -> DedupeResult:
    rows=connection.execute(f"SELECT dedupe_key,GROUP_CONCAT(person_id),COUNT(*) FROM person WHERE person_id IN ({','.join('?' for _ in person_ids)}) GROUP BY dedupe_key",tuple(person_ids)).fetchall()
    duplicates=conflicts=0
    with connection:
        for key,ids,count in rows:
            if count < 2: continue
            duplicates += count-1
            obs=connection.execute("SELECT COUNT(DISTINCT value) FROM source_observation WHERE entity_type='person' AND entity_id IN (SELECT value FROM json_each(?))",(json.dumps(ids.split(',')),)).fetchone()[0]
            if obs > 1:
                observation_ids=tuple(row[0] for row in connection.execute("SELECT observation_id FROM source_observation WHERE entity_type='person' AND entity_id IN (SELECT value FROM json_each(?)) ORDER BY observation_id",(json.dumps(ids.split(',')),)))
                connection.execute("INSERT INTO merge_review(review_id,entity_type,candidate_ids,observation_ids,reason,state,decided_by,decided_at) VALUES(?,?,?,?,?,'open',NULL,NULL)",(str(uuid.uuid4()),"person",json.dumps(ids.split(',')),json.dumps(observation_ids),"conflicting_observations")); conflicts += 1
    return DedupeResult(len(rows),duplicates,conflicts)

def ensure_fit_score_v1(connection: sqlite3.Connection, at: str) -> str:
    canonical=json.dumps(FIT_SCORE_V1_RULES,sort_keys=True,separators=(",",":")); digest=hashlib.sha256(canonical.encode()).hexdigest()
    row=connection.execute("SELECT fit_score_version_id FROM fit_score_version WHERE rule_hash=?",(digest,)).fetchone()
    if row: return row[0]
    version_id=str(uuid.uuid4()); insert_fit_score_version(connection,FitScoreVersion(version_id,"fit-v1",canonical,digest,at,at)); return version_id

def score_person(connection: sqlite3.Connection, campaign_id: str, person_id: str, observations: Mapping[str,object], at: str) -> int:
    version=ensure_fit_score_v1(connection,at); score=sum(weight for name,weight in FIT_SCORE_V1_RULES.items() if observations.get(name) is True)
    components=json.dumps({name:bool(observations.get(name)) for name in FIT_SCORE_V1_RULES},sort_keys=True,separators=(",",":"))
    connection.execute("INSERT INTO fit_score(fit_score_id,campaign_id,person_id,fit_score_version_id,score,components,scored_at) VALUES(?,?,?,?,?,?,?)",(str(uuid.uuid4()),campaign_id,person_id,version,score,components,at)); return score

def write_eligibility(connection, campaign_id, person_id, policy, score_version_id, at):
    decision=decide_eligibility(connection,campaign_id,person_id,policy,score_version_id,at)
    connection.execute("INSERT INTO eligibility_decision(decision_id,campaign_id,person_id,rule_version,fit_score_version_id,outcome,failed_predicate_ids,approximate_predicate_ids,decided_at,override_id) VALUES(?,?,?,?,?,?,?,?,?,?)",tuple(decision.__dict__.values()))
    return decision
```
