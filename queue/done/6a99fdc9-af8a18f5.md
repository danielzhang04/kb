@dataclass(frozen=True)
class SendResult:
    state: str
    code: str

def _jitter(logical_key: str) -> int:
    return 480 + int(hashlib.sha256(logical_key.encode()).hexdigest()[:8],16) % 241

def stagger_deliveries(connection, campaign_id, batch_hash, now):
    rows=connection.execute(
        "SELECT delivery_id,logical_key,scheduled_at FROM delivery WHERE campaign_id=? "
        "AND step=0 AND state='reserved' ORDER BY logical_key",(campaign_id,)
    ).fetchall()
    cursor=None; values=[]
    for delivery_id,logical_key,start_text in rows[:25]:
        start=datetime.fromisoformat(start_text)
        candidate=start if cursor is None else cursor+timedelta(seconds=_jitter(logical_key))
        approval=connection.execute(
            "SELECT send_window FROM approval WHERE campaign_id=? AND revision_hash="
            "(SELECT revision_hash FROM delivery WHERE delivery_id=?)",
            (campaign_id,delivery_id),
        ).fetchone()
        window=json.loads(approval[0])
        if candidate > datetime.fromisoformat(window["end"]):
            break
        connection.execute(
            "UPDATE delivery SET scheduled_at=? WHERE delivery_id=? AND state='reserved'",
            (candidate.isoformat(),delivery_id),
        )
        cursor=candidate; values.append(candidate)
    return tuple(values)

def _applicable(connection, delivery_id, now):

        "a.contact_id,a.mailbox_id,a.tier,a.send_window,a.expires_at,a.nonce,"

        "JOIN campaign c ON c.campaign_id=d.campaign_id "
        "JOIN enrollment e ON e.enrollment_id=d.enrollment_id "
        "JOIN contact_point cp ON cp.contact_id=d.contact_id AND cp.state='valid' "
        "JOIN approval a ON a.campaign_id=d.campaign_id AND a.policy_hash=c.policy_hash "
        "AND a.revision_hash=d.revision_hash AND a.contact_id=d.contact_id "
        "AND a.mailbox_id=d.mailbox_id "
        "WHERE d.delivery_id=? AND d.step=0 AND d.state='reserved' "

def circuit_open(connection, now):
    warning=connection.execute(
        "SELECT 1 FROM suppression WHERE reason='google_warning' AND released_at IS NULL"
    ).fetchone()
    bounces=connection.execute(
        "SELECT count(*) FROM suppression WHERE reason='bounce' AND released_at IS NULL "
        "AND substr(created_at,1,10)=?",(now.date().isoformat(),)
    ).fetchone()[0]
    return warning is not None or bounces >= 2

def queue_due_t1(connection, now):
    if circuit_open(connection,now):
        breaker_id=hashlib.sha256(
            f"bounce-breaker:{now.date().isoformat()}".encode()
        ).hexdigest()
        with connection:
            connection.execute(
                "INSERT OR IGNORE INTO suppression VALUES(?,?,?,?,?,?,?)",
                (breaker_id,"global","all","bounce",now.isoformat(),"executor",None),
            )
        return 0
    if connection.execute(
        "SELECT 1 FROM suppression WHERE scope='global' AND released_at IS NULL"
    ).fetchone():
        return 0
    queued=0
    for (delivery_id,) in connection.execute(
        "SELECT delivery_id FROM delivery WHERE step=0 AND state='reserved' "
        "AND scheduled_at<=? ORDER BY scheduled_at,logical_key",(now.isoformat(),)
    ):
        row=_applicable(connection,delivery_id,now)
        if row is None:
            continue
        approval_id=row[0]
        request_id=hashlib.sha256(f"gmail_send:{delivery_id}".encode()).hexdigest()
        payload=json.dumps({"delivery_id":delivery_id},sort_keys=True,separators=(",",":"))

            "request_id,caller,operation,payload,policy_hash,approval_id,created_at,state"

    seed=hashlib.sha256(f"{batch_hash}:{logical_key}".encode()).hexdigest()
    return 480 + int(seed[:8],16) % 241

def stagger_deliveries(connection,campaign_id,batch_hash,now):
    campaign=connection.execute(
        "SELECT timezone,daily_cap,hourly_cap FROM campaign WHERE campaign_id=?",
        (campaign_id,),
    ).fetchone()
    zone=ZoneInfo(campaign[0])
    local_now=now.astimezone(zone)
    rows=connection.execute(
        "SELECT d.delivery_id,d.logical_key,json_extract(a.send_window,'$.start'),"
        "json_extract(a.send_window,'$.end') FROM delivery d JOIN approval a "
        "ON a.campaign_id=d.campaign_id AND a.revision_hash=d.revision_hash "
        "AND a.contact_id=d.contact_id AND a.mailbox_id=d.mailbox_id "
        "WHERE d.campaign_id=? AND d.step=0 AND d.state='reserved' "
        "AND a.consumed_at IS NULL ORDER BY d.logical_key",(campaign_id,),
    ).fetchall()
    values=[]; cursor=None
    for delivery_id,logical_key,start_text,end_text in rows[:min(25,campaign[1])]:
        start=datetime.fromisoformat(start_text); end=datetime.fromisoformat(end_text)
        if local_now.date() >= end.astimezone(zone).date() or now >= start:
            connection.execute(
                "UPDATE enrollment SET block_reason='machine_unavailable' WHERE enrollment_id="
                "(SELECT enrollment_id FROM delivery WHERE delivery_id=?)",(delivery_id,),
