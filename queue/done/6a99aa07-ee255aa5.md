from dataclasses import dataclass
import os, random
from pathlib import Path
from urllib.parse import urlsplit

@dataclass(frozen=True)
class BrowserPolicy: allowed_hosts: tuple[str,...]; rolling_cap: int=40; session_seconds: int=1800
def check_navigation(url,policy):
    parsed=urlsplit(url)
    if parsed.scheme=="file" and os.environ.get("KB_PROSPECTING_NO_NETWORK")=="1": return
    if parsed.scheme!="https" or parsed.hostname not in policy.allowed_hosts: raise ValueError("browser_domain_blocked")
    if parsed.scheme=="https" and not parsed.path.startswith(("/search/results/people/","/in/","/company/")): raise ValueError("browser_action_blocked")
def next_delay(random_source): return random_source.randint(45,120)
def check_action(action):
    allowed={"search_results","profile_page","company_page","close","wait"}
    if action not in allowed: raise ValueError(f"browser_action_blocked:{action}")
def detect_stop(html):
    lowered=html.casefold()
    return next((m.replace(" ","_") for m in ("captcha","checkpoint","rate warning","login challenge","unexpected modal") if m in lowered),None)

def open_persistent_page(playwright,user_data_dir):
    context=playwright.chromium.launch_persistent_context(str(user_data_dir),channel="chrome",headless=False)
    return context.pages[0] if context.pages else context.new_page()

def dedicated_user_data_dir(environ=None):
    values=os.environ if environ is None else environ
    root=values.get("LOCALAPPDATA")
    if not root: raise RuntimeError("localappdata_unavailable")
    return Path(root)/"kb-outreach-chrome"

from html.parser import HTMLParser

class _Fields(HTMLParser):
    def __init__(self): super().__init__(); self.values={}; self.current=None
    def handle_starttag(self,tag,attrs):
        data=dict(attrs); self.current=data.get("data-field")
        if self.current and "href" in data: self.values[self.current]=data["href"]
    def handle_data(self,data):
        if self.current and data.strip(): self.values[self.current]=data.strip()
    def handle_endtag(self,tag): self.current=None
def _parse(html): p=_Fields(); p.feed(html); return p.values
def parse_profile(html):
    allowed={"first_name","full_name","title","location","linkedin_url","company_name","company_linkedin_url"}; return {k:v for k,v in _parse(html).items() if k in allowed}
def parse_company(html):
    allowed={"name","website_url","linkedin_url","industry","location","one_line_summary"}; return {k:v for k,v in _parse(html).items() if k in allowed}
def parse_search_results(html):
    rows=[]
    for part in html.split('data-result="')[1:]:
        values=_parse(part.split("</article>",1)[0]); rows.append({k:v for k,v in values.items() if k in {"full_name","title","linkedin_url","company_name"}})
    return tuple(rows)

from __future__ import annotations
from datetime import timedelta
import hashlib, json, sqlite3, uuid
from scripts.prospecting.browser_guard import BrowserPolicy,check_navigation,detect_stop,next_delay
from scripts.prospecting.lanes import LaneBatch,LanePlan,YieldEstimate,capabilities_for

class LinkedInBudget:
    def __init__(self,connection,clock): self.connection,self.clock,self.started=connection,clock,None
    def start_session(self): self.started=self.clock.now()
    def may_navigate(self): return self.started is None or (self.clock.now()-self.started).total_seconds()<1800
    def cap_reached(self):
        cutoff=(self.clock.now()-timedelta(hours=24)).isoformat()
        return self.connection.execute("SELECT COUNT(*) FROM audit WHERE action='linkedin_profile_load_reserved' AND at>?",(cutoff,)).fetchone()[0]>=40
    def reserve_navigation(self,url):
        cutoff=(self.clock.now()-timedelta(hours=24)).isoformat()
        self.connection.execute("BEGIN IMMEDIATE")
        count=self.connection.execute("SELECT COUNT(*) FROM audit WHERE action='linkedin_profile_load_reserved' AND at>?",(cutoff,)).fetchone()[0]
        if count>=40:
            self.connection.rollback(); return None
        load_id=str(uuid.uuid4()); digest=hashlib.sha256(url.encode()).hexdigest()
        self.connection.execute("INSERT INTO audit(event_id,actor,action,entity_type,entity_id,at,before_hash,after_hash,reason) VALUES(?,?,?,?,?,?,?,?,?)",(load_id,"executor","linkedin_profile_load_reserved","finder_run",load_id,self.clock.now().isoformat(),None,digest,"cap_reserved"))
        self.connection.commit(); return load_id
    def mark_success(self,load_id,url):
        digest=hashlib.sha256(url.encode()).hexdigest()
        self.connection.execute("INSERT INTO audit(event_id,actor,action,entity_type,entity_id,at,before_hash,after_hash,reason) VALUES(?,?,?,?,?,?,?,?,?)",(str(uuid.uuid4()),"executor","linkedin_profile_load_success","finder_run",load_id,self.clock.now().isoformat(),None,digest,"visible_page")); self.connection.commit()

class LinkedInAssistedLane:
    name="linkedin_assisted"; capability_version="linkedin-v1"
    def __init__(self,page_factory,clock,connection,rng,sleeper,on_stop=lambda reason:None): self.page_factory,self.clock,self.connection,self.rng,self.sleeper,self.on_stop=page_factory,clock,connection,rng,sleeper,on_stop
    def capabilities(self):
        return capabilities_for(self.name,self.capability_version)
    def plan(self,target_policy):
        if self.name not in target_policy.lanes: raise ValueError("linkedin_lane_disabled")
        return LanePlan(self.name,self.capability_version,self.capabilities(),YieldEstimate(5,15,40),40)
    def dry_run(self,plan,cursor,now): return tuple({"ordinal":i+1,"earliest_offset_seconds":45*i} for i in range(plan.limit))
    def run(self,plan,cursor): return LaneBatch((),cursor.cursor if cursor else None,0,0,False,None)
    def run_live(self,urls):
        budget=LinkedInBudget(self.connection,self.clock); budget.start_session(); page=self.page_factory(); elapsed=0
        for index,url in enumerate(urls):
            if budget.cap_reached(): page.close(); return "cap_reached"
            if not budget.may_navigate() or elapsed>=1800: page.close(); return "session_limit"
            load_id=budget.reserve_navigation(url)
            if load_id is None: page.close(); return "cap_reached"
            check_navigation(url,BrowserPolicy(("www.linkedin.com",))); page.goto(url)
            reason=detect_stop(page.content())
            if reason:
                page.close(); self.on_stop(reason); return reason
            budget.mark_success(load_id,url)
            if index+1<len(urls):
                delay=next_delay(self.rng)
                self.sleeper(delay); elapsed+=delay
        page.close(); return "exhausted"
