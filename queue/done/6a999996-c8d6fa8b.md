from typing import Literal

ContactState = Literal["valid","invalid","risky","catch_all","role","stale","not_found","error"]

@dataclass(frozen=True)
class EmailResult:
    state: ContactState; email_ref: str | None; confidence: float; at: str
    provider: str; credits: int; raw_response_ref: int | None = None

class EmailFinder(Protocol):
    def max_cost(self, operation: str) -> int: ...
    def find_work_email(self, person_ref: str, company_ref: str, idempotency_key: str) -> EmailResult: ...
    def verify_work_email(self, email_ref: str, idempotency_key: str) -> EmailResult: ...

class ExecutorEmailGateway:
    def __init__(self,executor,campaign_id): self.executor,self.campaign_id=executor,campaign_id
    def execute(self,adapter,operation,payload,key=None,*,request_id=None,person_id=None,now="2026-09-03T00:00:00Z"):
        request_id=request_id or key
        person_id=person_id or payload.get("person_ref")
        if not request_id or not person_id: raise ValueError("email_gateway_opaque_refs_required")
        class Bridge:
            max_retries=adapter.max_retries
            timeout_seconds=adapter.timeout_seconds
            def max_cost(self,op): return adapter.max_cost(op)
            def call(self,op,data,key):
                if op=="find": return adapter.find_work_email(data["person_ref"],data["company_ref"],key)
                if op=="verify": return adapter.verify_work_email(data["email_ref"],key)
                raise ValueError("unknown_email_operation")
        result=_execute_reserved(self.executor.connection,self.campaign_id,request_id,person_id,
            adapter.provider,operation,payload,request_id,Bridge(),now)
        if isinstance(result,EmailResult): return result
        return EmailResult(result.state,None,0.0,now,adapter.provider,result.credits,result.raw_response_ref)

class HunterEmailFinder:
    provider="hunter"
    timeout_seconds=20; max_retries=2
    def __init__(self, fixture_dir=Path("orgs/prospecting/fixtures/vendor/hunter"), transport=None): self.fixture_dir,self.transport,self._results=fixture_dir,transport,{}

        cache_key=(name,key)
        if cache_key in self._results: return self._results[cache_key]
        if os.environ.get("KB_PROSPECTING_NO_NETWORK")=="1": raw=json.loads((self.fixture_dir/name).read_text(encoding="utf-8"))
        elif self.transport is None: raise RuntimeError("hunter_transport_unavailable")

\# scripts/prospecting/bakeoff.py
from __future__ import annotations
from dataclasses import dataclass
import math, random
from typing import Mapping, Sequence

@dataclass(frozen=True)
class BakeoffCase:
    person_id: str; company_id: str; known_email_hash: str; observed_bounce_class: str

@dataclass(frozen=True)
class BakeoffMetric:
    provider: str; attempts: int; matches: int; wrong_person: int; not_found: int
    bounce_confusion: Mapping[str,int]; credits: int; cost_per_match: float | None
    match_interval: tuple[float,float]; wrong_person_interval: tuple[float,float]

def wilson_interval(successes: int, total: int):
    if total == 0: return (0.0,0.0)
    z=1.959963984540054; p=successes/total; d=1+z*z/total
    center=(p+z*z/(2*total))/d; margin=z*math.sqrt((p*(1-p)+z*z/(4*total))/total)/d
    return (center-margin,center+margin)

def run_bakeoff(cases: Sequence[BakeoffCase], providers: Sequence[object], gateway, seed: int):
    ordered=list(cases); random.Random(seed).shuffle(ordered); output=[]
    for provider in providers:
        matches=wrong=missing=credits=0; confusion={}
        for case in ordered:
            found=gateway.execute(provider,"find",{"person_ref":case.person_id,"company_ref":case.company_id},f"{seed}:{case.person_id}:find")
            credits += found.credits
            if found.email_ref is None: missing += 1; continue
            import hashlib
            if hashlib.sha256(found.email_ref.encode()).hexdigest()==case.known_email_hash: matches += 1
            else: wrong += 1
            verified=gateway.execute(provider,"verify",{"email_ref":found.email_ref,"person_ref":case.person_id},f"{seed}:{case.person_id}:verify"); credits += verified.credits
            key=f"{case.observed_bounce_class}->{verified.state}"; confusion[key]=confusion.get(key,0)+1
        total=len(ordered)
        output.append(BakeoffMetric(provider.provider,total,matches,wrong,missing,confusion,credits,None if matches==0 else credits/matches,wilson_interval(matches,total),wilson_interval(wrong,total)))
    return tuple(output)
