```dart
Future<PhoneBridgeOpenResult> openRecord(
  PhoneBridgeNotificationRecord record,
) async {
  if (opening.value) return PhoneBridgeOpenResult.failed;
  opening.value = true;
  try {
    await markRead(record);
    final opened = await _routeOpener.open(record.item);

    if (opened.outcome == PhoneBridgeRouteOpenOutcome.openedBrowser) {
      _clearIssue(PhoneBridgePanelIssueCode.chromeLaunchFailed);
    }
    if (opened.outcome == PhoneBridgeRouteOpenOutcome.openedPhoneLink) {
      _clearIssue(PhoneBridgePanelIssueCode.phoneLinkLaunchFailed);
    }

    switch (opened.reason) {
      case PhoneBridgeRouteOpenReason.none:
        break;
      case PhoneBridgeRouteOpenReason.routeResolutionFailed:
        _setIssue(
          const PhoneBridgePanelIssue(
            code: PhoneBridgePanelIssueCode.routeResolutionFailed,
            message:
                'The notification route was unavailable. Trying Phone Link instead.',
            canRetry: false,
          ),
        );
      case PhoneBridgeRouteOpenReason.browserLaunchFailed:
        _setIssue(
          const PhoneBridgePanelIssue(
            code: PhoneBridgePanelIssueCode.chromeLaunchFailed,
            message: 'The browser route failed. Trying Phone Link instead.',
            canRetry: false,
          ),
        );
      case PhoneBridgeRouteOpenReason.phoneLinkLaunchFailed:
        _setIssue(
          const PhoneBridgePanelIssue(
            code: PhoneBridgePanelIssueCode.phoneLinkLaunchFailed,
            message: 'Phone Link could not be opened.',
            canRetry: false,
          ),
        );
      case PhoneBridgeRouteOpenReason.openBubblesLaunchFailed:
        _setIssue(
          const PhoneBridgePanelIssue(
            code: PhoneBridgePanelIssueCode.routeResolutionFailed,
            message: 'OpenBubbles could not open this conversation.',
            canRetry: false,
          ),
        );
    }

    final result = switch (opened.outcome) {
      PhoneBridgeRouteOpenOutcome.openedBrowser =>
        PhoneBridgeOpenResult.chromeOpened,
      PhoneBridgeRouteOpenOutcome.openedPhoneLink
          when opened.reason == PhoneBridgeRouteOpenReason.none =>
        PhoneBridgeOpenResult.phoneLinkOpened,
      PhoneBridgeRouteOpenOutcome.openedPhoneLink =>
        PhoneBridgeOpenResult.phoneLinkFallbackOpened,
      PhoneBridgeRouteOpenOutcome.openedOpenBubbles =>
        PhoneBridgeOpenResult.openBubblesOpened,
      PhoneBridgeRouteOpenOutcome.failed => PhoneBridgeOpenResult.failed,
    };
    lastOpenResult.value = result;
    return result;
  } finally {
    opening.value = false;
  }
}
```
