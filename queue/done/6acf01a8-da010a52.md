\## Proposed scanner

This Python 3 stdlib scanner uses extended Windows paths, never descends through reparse points, reports errors in-band, projects destination paths, and returns an opaque continuation cursor. Each batch’s byte total covers only that batch; aggregate batches after reaching `complete: true`.

```python
\# scan_tree.py
import argparse, base64, json, os, stat

def longpath(path):
    path = os.path.abspath(path)
    if path.startswith("\\\\?\\"):
        return path
    if path.startswith("\\\\"):
        return "\\\\?\\UNC\\" + path[2:]
    return "\\\\?\\" + path

def display(path):
    if path.startswith("\\\\?\\UNC\\"):
        return "\\\\" + path[8:]
    if path.startswith("\\\\?\\"):
        return path[4:]
    return path

def encode_cursor(source, destination, pending):
    state = {"v": 1, "source": source, "destination": destination,
             "pending": pending}
    raw = json.dumps(state, separators=(",", ":")).encode()
    return base64.urlsafe_b64encode(raw).decode()

def decode_cursor(value, source, destination):
    state = json.loads(base64.urlsafe_b64decode(value.encode()))
    if (state.get("v"), state.get("source"), state.get("destination")) != (
            1, source, destination):
        raise SystemExit("cursor does not match source/destination")
    return state["pending"]

ap = argparse.ArgumentParser()
ap.add_argument("source")
ap.add_argument("destination")
ap.add_argument("--limit", type=int, default=5000)
ap.add_argument("--cursor")
args = ap.parse_args()

source = os.path.abspath(args.source)
destination = os.path.abspath(args.destination)
pending = (decode_cursor(args.cursor, source, destination)
           if args.cursor else [longpath(source)])

records, errors = [], []
logical_bytes = 0
max_source_length = max_destination_length = 0

while pending and len(records) < args.limit:
    path = pending.pop()
    shown = display(path)

    try:
        st = os.stat(path, follow_symlinks=False)
    except OSError as exc:
        errors.append({"path": shown, "operation": "lstat",
                       "error": repr(exc)})
        continue

    rel = os.path.relpath(shown, source)
    projected = destination if rel == "." else os.path.join(destination, rel)
    attrs = getattr(st, "st_file_attributes", 0)
    is_reparse = bool(attrs & getattr(
        stat, "FILE_ATTRIBUTE_REPARSE_POINT", 0x400))

    if stat.S_ISDIR(st.st_mode):
        kind = "directory"
    elif stat.S_ISREG(st.st_mode):
        kind = "file"
        logical_bytes += st.st_size
    else:
        kind = "other"

    record = {
        "relative_path": rel,
        "kind": kind,
        "logical_bytes": st.st_size if kind == "file" else 0,
        "attributes": attrs,
        "reparse": is_reparse,
        "reparse_tag": getattr(st, "st_reparse_tag", None),
        "mtime_ns": st.st_mtime_ns,
        "ctime_ns": st.st_ctime_ns,
        "file_id": st.st_ino,
        "source_length": len(shown),
        "projected_path": projected,
        "projected_length": len(projected),
    }

    if is_reparse:
        try:
            record["link_target"] = os.readlink(path)
        except OSError as exc:
            errors.append({"path": shown, "operation": "readlink",
                           "error": repr(exc)})

    records.append(record)
    max_source_length = max(max_source_length, len(shown))
    max_destination_length = max(max_destination_length, len(projected))

    if kind == "directory" and not is_reparse:
        try:
            with os.scandir(path) as entries:
                children = sorted(
                    (os.path.join(path, entry.name) for entry in entries),
                    reverse=True)
            pending.extend(children)
        except OSError as exc:
            errors.append({"path": shown, "operation": "scandir",
                           "error": repr(exc)})

result = {
    "source": source,
    "destination": destination,
    "records": records,
    "errors": errors,
    "batch_entries": len(records),
    "batch_logical_bytes": logical_bytes,
    "batch_max_source_length": max_source_length,
    "batch_max_projected_length": max_destination_length,
    "complete": not pending,
    "next_cursor": (None if not pending else
                    encode_cursor(source, destination, pending)),
}
print(json.dumps(result, indent=2))
```

\## Adversarial race review

The main failure mode is treating a successful scan as a lock:

- A file can change after hashing but before capture.
- A same-sized rewrite can evade size/timestamp-only comparison.
- A directory can gain a file after its enumeration.
- A link can be swapped for a directory between type check and traversal.
- The destination can be created after the collision check.
- Antivirus, sync, editor, or indexing software may write despite a human freeze.
- A resumable cursor can silently combine different filesystem generations.
- Restore verification against “latest” can validate the wrong capture.

Therefore, only the frozen B→capture→C equality establishes the checkpoint. Use content hashes, compare error sets, identify snapshots by explicit capture ID, recheck destination absence immediately before rename, and invalidate the batch on any unexplained delta.
