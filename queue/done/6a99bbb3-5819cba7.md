def verify_p1_contract(
    connection: sqlite3.Connection,
    store_module: ModuleType,
    schema_path: Path,
) -> tuple[str, ...]:
    """Return stable mismatch codes; an empty tuple is the only usable result."""
    errors: list[str] = []
    if connection.execute("PRAGMA journal_mode").fetchone()[0].lower() != "wal":
        errors.append("pragma:journal_mode")
    if connection.execute("PRAGMA foreign_keys").fetchone()[0] != 1:
        errors.append("pragma:foreign_keys")
    tables = {
        row[0] for row in connection.execute(
            "SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%'"
        )
    }
    if tables != EXPECTED_TABLES:
        errors.append("tables:inventory")
    for name, expected in EXPECTED_SIGNATURES.items():
        function = getattr(store_module, name, None)
        if function is None:
            errors.append(f"missing_function:{name}")
            continue
        actual = tuple(inspect.signature(function).parameters)
        if actual != expected:
            errors.append(f"signature:{name}")
    for table, expected in EXPECTED_COLUMNS.items():
        actual = tuple(row[1] for row in connection.execute(f"PRAGMA table_info({table})"))
        if actual != expected:
            errors.append(f"columns:{table}")
    for (table, column), expected in EXPECTED_ENUMS.items():
        row = connection.execute(
            "SELECT sql FROM sqlite_master WHERE type='table' AND name=?", (table,)
        ).fetchone()
        table_sql = "" if row is None else re.sub(r"\s+", " ", row[0]).lower()
        matches = re.findall(
            rf"\b{re.escape(column.lower())}\b\s+in\s*\(([^)]*)\)", table_sql
        )
        actual = frozenset(re.findall(r"'([^']+)'", matches[0])) if len(matches) == 1 else frozenset()
        if actual != expected:
            errors.append(f"enum:{table}.{column}")
    for table, expected in EXPECTED_FOREIGN_KEYS.items():
        actual = frozenset(
            (row[3], row[2], row[4])
            for row in connection.execute(f"PRAGMA foreign_key_list({table})")
        )
        if actual != expected:
            errors.append(f"foreign_keys:{table}")
    for table, required in EXPECTED_UNIQUES.items():
        actual: set[tuple[str, ...]] = set()
        for index in connection.execute(f"PRAGMA index_list({table})"):
            if index[2]:
                actual.add(tuple(row[2] for row in connection.execute(f"PRAGMA index_info({index[1]})")))
        primary = tuple(row[1] for row in connection.execute(f"PRAGMA table_info({table})") if row[5])
        if primary:
            actual.add(primary)
        if frozenset(actual) != required:
            errors.append(f"unique:{table}")
    for table, index_name, columns, unique in EXPECTED_INDEXES:
        rows = {row[1]: bool(row[2]) for row in connection.execute(f"PRAGMA index_list({table})")}
        actual_columns = tuple(
            row[2] for row in connection.execute(f"PRAGMA index_info({index_name})")
        )
        if rows.get(index_name) is not unique or actual_columns != columns:
            errors.append(f"index:{table}.{index_name}")
    schema = re.sub(r"\s+", " ", schema_path.read_text(encoding="utf-8").lower())
    if not all(token in schema for token in (
        "create trigger", "approval", "content_kind", "reply_template", "revision_hash", "raise",
    )):
        errors.append("trigger:approval_resolution")
    if not all(token in schema for token in (
        "create trigger", "audit", "before update", "before delete", "raise",
    )):
        errors.append("trigger:audit_append_only")
    return tuple(sorted(errors))
