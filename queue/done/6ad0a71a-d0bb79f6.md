def sha256_file(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        while block := f.read(1024 * 1024):
            h.update(block)
    return h.hexdigest()

def lexical_under(path: Path, parent: Path) -> bool:
    try:
        return (
            os.path.commonpath(
                [os.path.normcase(os.path.abspath(path)),
                 os.path.normcase(os.path.abspath(parent))]
            )
            == os.path.normcase(os.path.abspath(parent))
        )
    except ValueError:
        return False


def no_reparse_existing_chain(path: Path) -> None:
    absolute = Path(os.path.abspath(path))
    parts = absolute.parts
    current = Path(parts[0])
    for part in parts[1:]:
        current /= part
        if not os.path.lexists(current):
            break
        st = os.lstat(current)
        if getattr(st, "st_file_attributes", 0) & REPARSE:
            raise Refusal(f"reparse ancestor refused: {current}")

def is_reparse(st: os.stat_result) -> bool:

def snapshot(root: Path) -> dict[str, dict[str, Any]]:
    if not root.is_dir():
        raise Refusal(f"snapshot root missing: {root}")
    out: dict[str, dict[str, Any]] = {}

    def visit(directory: Path) -> None:
        with os.scandir(directory) as iterator:
            entries = sorted(iterator, key=lambda e: e.name)
        for entry in entries:
            st = entry.stat(follow_symlinks=False)
            if is_reparse(st) or entry.is_symlink():
                raise Refusal(f"reparse entry refused: {entry.path}")
            path = Path(entry.path)
            rel = path.relative_to(root).as_posix()
            if entry.is_dir(follow_symlinks=False):
                out[rel] = {"kind": "dir", "file_id": st.st_ino}
                visit(path)
            elif entry.is_file(follow_symlinks=False):
                out[rel] = {
                    "kind": "file",
                    "size": st.st_size,
                    "sha256": sha256_file(path),
                    "file_id": st.st_ino,
                }
            else:
                raise Refusal(f"unsupported entry kind: {path}")

    visit(root)
    return out


def portable_manifest(snap: dict[str, dict[str, Any]]) -> dict[str, Any]:

        entries.append(record)
    return {"version": 1, "coverage": "regular-files-and-directories-bytes-only",

def validate_manifest_schema(manifest: dict[str, Any]) -> None:

def run_bounded(
    argv: list[str], cwd: Path, env: dict[str, str], logs: Path,
    expected: set[int] = {0},
) -> tuple[int, bytes, bytes]:
    global command_counter
    command_counter += 1
    stem = f"{command_counter:03d}"
    stdout_path = logs / f"{stem}.stdout"
    stderr_path = logs / f"{stem}.stderr"
    with stdout_path.open("xb") as out, stderr_path.open("xb") as err:
        proc = subprocess.Popen(
            argv, cwd=cwd, env=env, stdin=subprocess.DEVNULL,
            stdout=out, stderr=err, shell=False,
        )
        timed_out = False
        try:
            code = proc.wait(timeout=TIMEOUT)
        except subprocess.TimeoutExpired:
            timed_out = True
            proc.kill()
            code = proc.wait(timeout=5)
    stdout = stdout_path.read_bytes()
    stderr = stderr_path.read_bytes()
    watchdogs.append({
        "argv0": argv[0],
        "timeout": timed_out,
        "returncode": code,
        "stdout_bytes": len(stdout),
        "stderr_bytes": len(stderr),
    })
    if timed_out:
        raise Refusal(f"child timeout: {argv[0]}")
    if len(stdout) > MAX_OUTPUT or len(stderr) > MAX_OUTPUT:
        raise Refusal(f"child output limit exceeded: {argv[0]}")
    if code not in expected:
        raise Refusal(
            f"child exit {code}, expected {sorted(expected)}: "
            f"{stderr[:500].decode(errors='replace')}"
        )
    return code, stdout, stderr

def pointer_target(base: Path, text: str) -> Path:
    candidate = Path(text.strip())
    return Path(os.path.abspath(candidate if candidate.is_absolute() else base / candidate))


def scan_external_git_pointers(root: Path) -> list[dict[str, str]]:
    refusals: list[dict[str, str]] = []
    snap = snapshot(root)
    for rel, item in snap.items():
        if item["kind"] != "file":
            continue
        path = root.joinpath(*PurePosixPath(rel).parts)
        lower = rel.lower()
        spec: tuple[str, Path] | None = None
        if PurePosixPath(lower).name == ".git":
            data = path.read_bytes()
            if len(data) > MAX_POINTER:
                raise Refusal(f"oversized .git pointer: {rel}")
            line = data.decode("utf-8", "strict").strip()
            if not line.lower().startswith("gitdir:"):
                raise Refusal(f"malformed .git pointer: {rel}")
            spec = ("gitdir", path.parent)
            value = line.split(":", 1)[1]
        elif lower.endswith("/.git/commondir"):
            spec, value = ("commondir", path.parent), path.read_text("utf-8")
        elif lower.endswith("/.git/objects/info/alternates"):
            spec, value = ("alternates", path.parent), path.read_text("utf-8")
        elif "/.git/worktrees/" in lower and lower.endswith("/gitdir"):
            spec, value = ("worktree-gitdir", path.parent), path.read_text("utf-8")
        else:
            continue
        if path.stat().st_size > MAX_POINTER:
            raise Refusal(f"oversized pointer: {rel}")
        for line in value.splitlines() or [value]:
            target = pointer_target(spec[1], line)
            if not lexical_under(target, root):
                refusals.append({
                    "kind": spec[0], "pointer": rel,
                    "classification": "external-refused",
                })
    return refusals

    others = nul_paths(git_run(
        git, run_root, outer, env, logs,
        "ls-files", "-z", "--others", "--exclude-standard"
    ))
    ignored = nul_paths(git_run(
        git, run_root, outer, env, logs,
        "ls-files", "-z", "--others", "--ignored", "--exclude-standard"
    ))
    expected_blob = git_blob_oid(b"base\nstaged\n").encode()
    state_rows = [
        row for row in staged.split(b"\0")
        if row.endswith(b"\tstate.txt")
    ]
    if len(state_rows) != 1 or expected_blob not in state_rows[0]:
        raise Refusal("staged state.txt blob mismatch")
    if (outer / "state.txt").read_bytes() != b"base\nstaged\nunstaged\n":
        raise Refusal("unstaged state.txt bytes mismatch")
    if "untracked.txt" not in others or "ignored.txt" not in ignored:
        raise Refusal("untracked/ignored classification mismatch")

    git_run(git, run_root, outer, env, logs,
            "fsck", "--full", "--strict", "--no-progress", "--no-dangling")
    git_run(git, run_root, nested, env, logs,
            "ls-files", "-z", "--stage")
    git_run(git, run_root, nested, env, logs,
            "fsck", "--full", "--strict", "--no-progress", "--no-dangling")

    outer_shallow = (outer / ".git/shallow").read_text().strip()
    nested_shallow = (nested / ".git/shallow").read_text().strip()
    if (outer_shallow != expected["base_oid"]
            or nested_shallow != expected["nested_oid"]):
        raise Refusal("shallow markers were not preserved")

    results.update({
        "head": head_oid,
        "base_ref": base_oid,
        "local_only_commit_preserved": head_oid != base_oid,
        "stash_ref": stash_oid,
        "staged_state_blob": expected_blob.decode(),
        "unstaged_state_bytes_sha256": sha256_file(outer / "state.txt"),
        "untracked": sorted(others),
        "ignored": sorted(ignored),
        "shallow": {
            "outer": {
                "oid": outer_shallow,
                "classification": "preserved local shallow-history boundary; "
                                  "not proof of complete remote history",
            },
            "nested": {
                "oid": nested_shallow,
                "classification": "preserved local shallow-history boundary; "
                                  "not proof of complete remote history",
            },
        },
    })
    return results


def important_git_hashes(snap: dict[str, dict[str, Any]]) -> dict[str, str]:
    result = {}
    for rel, item in snap.items():
        marker = "/" + rel.lower()
        if item["kind"] != "file" or "/.git/" not in marker:
            continue
        tail = marker.split("/.git/", 1)[1]
        if (tail in {"index", "config", "shallow", "head"}
                or tail.startswith("refs/") or tail.startswith("logs/")):
            result[rel] = item["sha256"]
    return result


def ids_differ(
    left: dict[str, dict[str, Any]], right: dict[str, dict[str, Any]]
) -> str:
    files = [p for p, v in left.items() if v["kind"] == "file"]
    if any(not left[p]["file_id"] or not right[p]["file_id"] for p in files):
        return "UNKNOWN"
    return "PASS" if all(
        left[p]["file_id"] != right[p]["file_id"] for p in files
    ) else "FAIL"


def cold_main(mode: str, run_name: str, expected_script_hash: str) -> int:
    root = safe_run_root(run_name)
    if not root.is_dir():
        return 2
    if sha256_file(Path(__file__).absolute()) != expected_script_hash:
        return 2

    mapping = {
        "--cold": ("recovery", "restore-env"),
        "--cold-tamper": ("tampered-recovery", "tamper-restore-env"),
        "--cold-occupied": ("recovery", "occupied-restore-env"),
    }
    if mode not in mapping:
        return 2
    recovery_name, destination_name = mapping[mode]
    recovery = root / recovery_name
    destination = root / destination_name

    try:
        manifest, payload = load_valid_recovery(recovery)
        if os.path.lexists(destination):
            return 4
        restore_from_manifest(payload, destination, manifest, root)
        compare_to_manifest(destination, manifest)
        print(json.dumps({
            "status": "PASS",
            "mode": mode,
            "source_fixture_reads": 0,
            "recovery_root": recovery_name,
            "destination": destination_name,
        }, sort_keys=True))
        return 0
    except Refusal as exc:
        print(json.dumps({
            "status": "REFUSED",
            "mode": mode,
            "reason": str(exc),
            "destination_created": os.path.lexists(destination),
        }, sort_keys=True))
        return 3
