\## Goal

Ship the fail-closed T1 boundary: Daniel approves an exact first-touch batch through the existing
dashboard WebAuthn ceremony, the desktop materializes one approval per exact revision/contact pair,
and the executor releases those first touches in the next permitted morning window. Preserve P4's
refresh/recheck/CAS boundary, caps, deterministic delivery identity, and circuit breaker. Deliver
proposed cadence text and deployment tooling, but leave cadence registration, promotion, campaign
enablement, and the first real batch to Daniel.

\## Architecture

The VM may assemble IDs and display a PII-free summary, but it never receives names, addresses,
message bytes, credentials, or a send capability. The canonical approval scope is staged as the
work-order bytes of a normal kb approval card. The dashboard signs those committed bytes; the
desktop verifier imports the existing kb WebAuthn verifier, acts only on the verifier's pinned
card/result, and never parses public-key material itself.

A batch scope binds the campaign, policy, sorted exact revision/contact pairs, batch hash, mailbox,
T1 tier, permitted window, expiry, and a random parent nonce. Materialization is one transaction.
The first approval uses the parent nonce and later rows use deterministic child nonces derived from
the parent nonce plus the exact item. This preserves the P1 unique approval.nonce contract, makes
replaying the assertion fail the entire transaction, and leaves each send with a separately
consumable nonce. Each row's scope_hash is the canonical hash of its exact send scope.

The T1 executor is the only live-send path. It scans inbound first, refreshes the Gmail thread,
rechecks every bound field and all hooks, compare-and-set claims the delivery, and consumes the
matching approval in the same SQLite transaction immediately before the Gmail adapter call.
Follow-ups continue through P4's draft path. T2, T3, sales, and reply-template automation remain
fail-closed.

```python
\# scripts/prospecting/p6_contracts.py
from __future__ import annotations
import hashlib
import inspect
import json
import sqlite3
import subprocess
from pathlib import Path
from typing import Callable

PHASES = ("P1","P2","P3","P4","P5")
EXPECTED_TABLES = {
    "company","person","source_observation","employment","merge_review",
    "fit_score_version","fit_score","eligibility_decision","contact_point",
    "provider_attempt","credit_reservation","finder_run","finder_cursor",
    "source_snapshot","campaign","evidence","revision","reply_template","approval",
    "enrollment","delivery","inbound","reply_revision","exec_request","suppression",
    "relationship","audit",
}
EXPECTED_TRIGGERS = {"approval_resolution", "audit_append_only"}

def verify_prerequisites(
    root: Path, runner: Callable = subprocess.run, pending_ok: bool = False,
) -> dict[str, object]:
    hashes: dict[str,str] = {}
    pending: list[str] = []
    for phase in PHASES:
        argv = [
            "py","-3","-m","scripts.prospecting.gate",
            "--phase",phase,"--verify-recorded",
        ]
        result = runner(argv,cwd=root,text=True,capture_output=True,check=False)
        try:
            value = json.loads(result.stdout)
        except (TypeError,json.JSONDecodeError) as exc:
            raise RuntimeError(f"invalid_recorded_gate:{phase}") from exc
        if value.get("phase") != phase:
            raise RuntimeError(f"invalid_recorded_gate:{phase}")
        if value.get("recorded") == "absent":
            if pending_ok and phase != "P1":
                pending.append(phase)
                continue
            raise RuntimeError(f"recorded_gate_absent:{phase}")
        if result.returncode != 0 or value.get("recorded") != "matched" or value.get("passed") is not True:
            raise RuntimeError(f"recorded_gate_failed:{phase}")
        hashes[phase] = hashlib.sha256(result.stdout.encode()).hexdigest()
    return {"phases":list(PHASES),"record_sha256":hashes,"pending":pending}

def validate_schema(connection: sqlite3.Connection) -> tuple[str, ...]:
    objects = connection.execute(
        "SELECT type,name,sql FROM sqlite_master WHERE type IN ('table','index','trigger')"
    ).fetchall()
    tables = {name for kind,name,sql in objects if kind == "table" and not name.startswith("sqlite_")}
    triggers = {name for kind,name,sql in objects if kind == "trigger"}
    errors = [f"table:{name}" for name in sorted(EXPECTED_TABLES - tables)]
    errors += [f"trigger:{name}" for name in sorted(EXPECTED_TRIGGERS - triggers)]
    ddl = "\n".join(sql or "" for _,_,sql in objects)
    for token in (" CHECK "," UNIQUE","CREATE UNIQUE INDEX"):
        if token.lower() not in ddl.lower():
            errors.append(f"constraint:{token.strip().lower()}")
    if connection.execute("PRAGMA journal_mode").fetchone()[0].lower() != "wal":
        errors.append("pragma:journal_mode")
    if connection.execute("PRAGMA foreign_keys").fetchone()[0] != 1:
        errors.append("pragma:foreign_keys")
    return tuple(errors)

def validate_pre_commit_hook(root: Path) -> tuple[str, ...]:
    lines = (root / ".githooks/pre-commit").read_text(encoding="utf-8").splitlines()
    retained = lines[3:18]
    errors = []
    if len(retained) != 15 or not any("skill" in line.lower() for line in retained):
        errors.append("hook:retention")
    if not any("pii" in line.lower() for line in lines):
        errors.append("hook:pii_guard")
    return tuple(errors)

def validate_runtime_contracts() -> tuple[str, ...]:
    from scripts.prospecting.campaigner.cli import CampaignerService
    from scripts.prospecting.campaigner.release import release_due
    from scripts.prospecting.executor import validate_exec_request
    from scripts.prospecting.manager.bridge import run_campaigner_job
    checks = {
        "release_due": (release_due, ("context","delivery_id")),
        "scan": (CampaignerService.scan, ("self",)),
        "sweep": (CampaignerService.sweep, ("self",)),
        "status": (CampaignerService.status, ("self",)),
        "executor_validator": (validate_exec_request,("request","campaign_tier")),
        "bridge_wrapper": (run_campaigner_job,("job","mode","host","timeout")),
    }
    errors = []
    for name, (callable_, expected) in checks.items():
        names = tuple(inspect.signature(callable_).parameters)
        if names != expected:
            errors.append(f"{name}:signature:{','.join(names)}")
    return tuple(errors)
```
