\## Goal

Ship the bounded P5 control plane that turns one desktop-local plain-English ask into a typed, PII-free campaign policy; creates one local-outbox card per workflow stage; runs the five flat workflows through four least-privilege agents and the existing independent inspector; invokes the deterministic Windows desktop executor from a VM terminal through one opaque SSH envelope; and proves a resumable synthetic run without a model, live vendor, live Gmail send, live cadence registration, or coordination-branch write.

\## Architecture

The Windows desktop is the executor host. It owns SQLite, the isolated Chrome profile, Gmail and vendor credentials, ask text, all PII, policy compilation, and every deterministic domain CLI. The kb VM is the orchestrator. It owns agent/workflow declarations, PII-free cards, aggregate state, and the manager runner. The VM reaches the desktop only through the fixed Tailscale SSH bridge described in Task 6.

The orchestration graph is flat. `prospecting-manager` writes stage cards and consumes typed results; it never asks an agent to spawn another agent. Each producing stage is followed by the existing `inspector` where the workflow requires it. A failed inspection blocks downstream cards. The deliverable split preserves one table/write boundary, one permission surface, one memory shard, and one independent draft-eval suite per agent.

Raw asks are compiled on the desktop. The VM receives only an opaque `ask_ref`, canonical `target_policy`, canonical `campaign_policy`, opaque entity/run IDs, hashes, counts, states, grades, and failure codes. The terminal interface retains the required `--ask` spelling, but on the VM its value is an opaque desktop `ask_ref`; literal ask text is accepted only by `--local` on the desktop. This is the only interpretation consistent with the binding PII law, which forbids raw ask bytes in VM arguments and shell history.

P5 creates no live schedule. It verifies the two P4 cadence blocks, if present, remain inert (`standing_authority: false`). Promotion of declarations to the VM, blessing evals, cadence authorship on protected `main`, T3 sending, and dashboard launch remain human ceremonies.

from __future__ import annotations
from dataclasses import asdict, dataclass
import json
from pathlib import Path
from typing import Callable
from scripts.prospecting.manager.jobs import StageJob, write_card
from scripts.prospecting.pii_guard import assert_vm_safe

ALLOWED_RESULT = {"stage_id","state","ids","counts","hashes","failure_codes","attempt"}
HEX64 = __import__("re").compile(r"^[0-9a-f]{64}$")
@dataclass(frozen=True)
class RunReport:
    run_id: str; workflow: str; state: str; cards: int; retries: int; escalations: int; counts: dict[str,int]

class ManagerRunner:
    def __init__(self, workflow, outbox: Path, turn: Callable, inspector_turn: Callable): self.workflow,self.outbox,self.turn,self.inspector_turn=workflow,outbox,turn,inspector_turn
    def _valid(self, value, stage_id: str) -> dict:
        if not isinstance(value, dict) or set(value) != ALLOWED_RESULT or value.get("stage_id") != stage_id: raise ValueError("invalid_stage_result")
        if value.get("state") not in {"complete","failed"}: raise ValueError("invalid_stage_state")
        if not isinstance(value["ids"],list) or not all(isinstance(v,str) for v in value["ids"]): raise ValueError("invalid_ids")
        if not isinstance(value["hashes"],list) or not all(isinstance(v,str) and HEX64.fullmatch(v) for v in value["hashes"]): raise ValueError("invalid_hashes")
        if not isinstance(value["counts"],dict) or not all(isinstance(k,str) and type(v) is int and v >= 0 for k,v in value["counts"].items()): raise ValueError("invalid_counts")
        if not isinstance(value["failure_codes"],dict) or not all(isinstance(k,str) and type(v) is int and v >= 0 for k,v in value["failure_codes"].items()): raise ValueError("invalid_failure_codes")
        if type(value["attempt"]) is not int or value["attempt"] < 1: raise ValueError("invalid_attempt")
        assert_vm_safe(value, "stage_result"); return value
    def _checkpoint_path(self, run_id: str) -> Path: return self.outbox/f"{run_id}-checkpoint.json"
    def _save(self, run_id: str, value: dict) -> None:
        assert_vm_safe(value,"checkpoint"); self.outbox.mkdir(parents=True,exist_ok=True)
        target=self._checkpoint_path(run_id); temporary=target.with_suffix(".tmp")
        temporary.write_text(json.dumps(value,sort_keys=True,separators=(",",":")),encoding="utf-8"); temporary.replace(target)
    def _wake(self, run_id: str, policy_id: str, policy_hash: str, reason: str) -> None:
        write_card(StageJob(f"{run_id}-wake", "prospecting", run_id, "wake-me", "human", (),
          policy_id,policy_hash,(run_id,),(),{},("human_decision_required",reason)), self.outbox)
    def _park(self, run_id: str, policy_id: str, policy_hash: str, reason: str,
              completed: dict, cards: int, retries: int, escalations: int,
              totals: dict[str,int]) -> RunReport:
        self._wake(run_id,policy_id,policy_hash,reason); escalations+=1
        self._save(run_id,{"state":"parked","reason":reason,"completed":completed,"cards":cards,
                           "retries":retries,"escalations":escalations,"counts":totals})
        return RunReport(run_id,self.workflow.id,"parked",cards,retries,escalations,totals)
    def _inspect(self, stage, result: dict) -> dict:
        verdict = self.inspector_turn(stage, result)
        if not isinstance(verdict, dict) or set(verdict) != {"decision","grade"}: raise ValueError("invalid_inspector_result")
        if verdict["decision"] not in {"pass","retry","park"} or type(verdict["grade"]) is not int or not 0 <= verdict["grade"] <= 100: raise ValueError("invalid_inspector_decision")
        assert_vm_safe(verdict, "inspector_result"); return verdict
    def run(self, run_id: str, policy_id: str, policy_hash: str, input_ids: tuple[str,...],
            input_hashes: tuple[str,...], counts: dict[str,int]) -> RunReport:
        assert_vm_safe({"policy_id":policy_id,"policy_hash":policy_hash,"input_ids":input_ids,"input_hashes":input_hashes,"counts":counts}, "run_input")
        checkpoint=self._checkpoint_path(run_id)
        state=json.loads(checkpoint.read_text(encoding="utf-8")) if checkpoint.exists() else {"completed":{},"cards":0,"retries":0,"escalations":0,"counts":{}}
        if state.get("state") in {"complete","parked"}: return RunReport(run_id,self.workflow.id,state["state"],state["cards"],state["retries"],state["escalations"],state["counts"])
        completed=state["completed"]; cards=state["cards"]; retries=state["retries"]; escalations=state["escalations"]; totals=state["counts"]
        for stage in self.workflow.stages:
            if stage.id in completed: continue
            if any(dep not in completed for dep in stage.needs):
                return self._park(run_id,policy_id,policy_hash,"dependency_not_complete",completed,cards,retries,escalations,totals)
            deps=tuple(completed[dep]["card_id"] for dep in stage.needs)
            prior=[completed[dep]["result"] for dep in stage.needs]
            stage_ids=tuple(dict.fromkeys([*input_ids,*(item for result in prior for item in result["ids"])]))
            stage_hashes=tuple(dict.fromkeys([*input_hashes,*(item for result in prior for item in result["hashes"])]))
            card_id=f"{run_id}-{stage.id}-a1"
            job=StageJob(card_id,"prospecting",run_id,stage.id,stage.agent,deps,policy_id,policy_hash,stage_ids,stage_hashes,counts,("summary_schema_valid","no_pii"))
            write_card(job,self.outbox); cards+=1
            if stage.agent == "human": return self._park(run_id,policy_id,policy_hash,"human_gate",completed,cards,retries,escalations,totals)
            if stage.inspect:
                producer_id=stage.needs[0]; producer=next(item for item in self.workflow.stages if item.id == producer_id); producer_result=completed[producer_id]["result"]
                try: verdict=self._inspect(stage,producer_result)
                except Exception: return self._park(run_id,policy_id,policy_hash,"invalid_or_missing_result",completed,cards,retries,escalations,totals)
                if verdict["decision"] == "retry":
                    retries+=1; rework_id=f"{run_id}-{producer_id}-a2"
                    rework_job=StageJob(rework_id,"prospecting",run_id,producer_id,producer.agent,(card_id,),policy_id,policy_hash,stage_ids,stage_hashes,counts,("inspector_rework_once","summary_schema_valid","no_pii"))
                    write_card(rework_job,self.outbox); cards+=1
                    try: rework=self._valid(self.turn(producer,2,rework_job),producer_id)
                    except Exception: rework={"state":"failed"}
