function Get-ShaBytes([byte[]]$Bytes) {
    $sha = [Security.Cryptography.SHA256]::Create()
    try { [BitConverter]::ToString($sha.ComputeHash($Bytes)).Replace('-', '') }
    finally { $sha.Dispose() }
}

function Count-Ordinal([string]$Text, [string]$Needle) {
    if ($Needle.Length -eq 0) { throw 'empty count needle' }
    $count = 0
    $at = 0
    while ($true) {
        $p = $Text.IndexOf($Needle, $at, [StringComparison]::Ordinal)
        if ($p -lt 0) { break }
        $count++
        $at = $p + $Needle.Length
    }
    $count
}

function Splice-Once([string]$Text, [string]$Old, [string]$New, [string]$Label) {
    $count = Count-Ordinal $Text $Old
    if ($count -ne 1) { throw "$Label anchor count is $count, expected 1" }
    $p = $Text.IndexOf($Old, [StringComparison]::Ordinal)
    $result = $Text.Substring(0, $p) + $New + $Text.Substring($p + $Old.Length)
    if ((Count-Ordinal $result $New) -ne 1) {
        throw "$Label injected-span count is not exactly 1"

    if ($inverse -cne $Text) { throw "$Label inverse text proof failed" }
    $result
}

function Get-AssignmentLiteral($Ast, [string]$Name) {
    $hits = @($Ast.FindAll({
        param($node)
        if ($node -isnot [Management.Automation.Language.AssignmentStatementAst]) { return $false }
        if ($node.Left -isnot [Management.Automation.Language.VariableExpressionAst]) { return $false }
        $node.Left.VariablePath.UserPath -ceq $Name

    $childBytes = $Utf8.GetBytes($childText)
    $childSha = Get-ShaBytes $childBytes
    Write-BoundNew $ChildPath $childBytes
    if ((Get-FileSha $ChildPath) -cne $childSha) {
        throw 'fresh child readback hash mismatch'
    }

    $seedBytes = Read-Bounded $AcceptedManifestPath $MaxInputBytes
    if ((Get-ShaBytes $seedBytes) -cne $ExpectedAcceptedManifestSha) {
        throw 'accepted manifest changed between hash and bounded read'
    }
    $seedText = [Ait25ProvisionalCapture.JsonScan]::Validate(
        $seedBytes, 64, 8000000
    )
    [void](ConvertFrom-Json -InputObject $seedText)

    $definitions = @(
        [ordered]@{ name='wrong-external-digest'; expectedCategory='integrity'; expectedMessage='manifest SHA256 .* does not equal the externally pinned'; expectedLastStep='cli.bound'; mode='wrong-pin' },
        [ordered]@{ name='missing-manifest'; expectedCategory='integrity'; expectedMessage="cannot read 'manifest\.json'"; expectedLastStep='cli.bound'; mode='omit' },
        [ordered]@{ name='malformed-json'; expectedCategory='malformed'; expectedMessage='manifest refused before materialization: json:'; expectedLastStep='manifest.digest'; mode='raw-truncate' },
        [ordered]@{ name='duplicate-exact-root-key'; expectedCategory='malformed'; expectedMessage='duplicate or case-colliding property name'; expectedLastStep='manifest.digest'; mode='dup-exact' },
        [ordered]@{ name='duplicate-escaped-root-key'; expectedCategory='malformed'; expectedMessage='duplicate or case-colliding property name'; expectedLastStep='manifest.digest'; mode='dup-escaped' },
        [ordered]@{ name='duplicate-case-root-key'; expectedCategory='malformed'; expectedMessage='duplicate or case-colliding property name'; expectedLastStep='manifest.digest'; mode='dup-case' },
        [ordered]@{ name='nested-stream-duplicate-key'; expectedCategory='malformed'; expectedMessage='duplicate or case-colliding property name'; expectedLastStep='manifest.digest'; mode='dup-stream' },
        [ordered]@{ name='missing-required-key'; expectedCategory='malformed'; expectedMessage="manifest is missing field 'counts'"; expectedLastStep='manifest.digest'; mode='missing-key' },
        [ordered]@{ name='extra-unknown-key'; expectedCategory='malformed'; expectedMessage="manifest has unexpected field 'unexpectedA25n'"; expectedLastStep='manifest.digest'; mode='extra-key' },
        [ordered]@{ name='wrong-schema'; expectedCategory='malformed'; expectedMessage='manifest\.schema is not a valid string'; expectedLastStep='manifest.digest'; mode='wrong-schema' },
        [ordered]@{ name='traversal-path'; expectedCategory='malformed'; expectedMessage='traversal segment'; expectedLastStep='manifest.provenance'; mode='traversal' },
        [ordered]@{ name='case-colliding-entry-path'; expectedCategory='malformed'; expectedMessage='case-colliding manifest path'; expectedLastStep='manifest.provenance'; mode='path-case' },
        [ordered]@{ name='source-root-id-mismatch'; expectedCategory='malformed'; expectedMessage='provenance\.sourceRootId is not the root entry sourceId'; expectedLastStep='manifest.provenance'; mode='root-id' },
        [ordered]@{ name='repeated-source-id'; expectedCategory='malformed'; expectedMessage='duplicate manifest sourceId'; expectedLastStep='manifest.provenance'; mode='repeat-id' },
        [ordered]@{ name='contradictory-policy-text'; expectedCategory='unsupported'; expectedMessage='policy\.namedStreams is not this program''s exact policy text'; expectedLastStep='manifest.provenance'; mode='policy' },
        [ordered]@{ name='invalid-utc-date-wrapper'; expectedCategory='malformed'; expectedMessage='manifest\.createdUtc is not a canonical UTC round-trip'; expectedLastStep='manifest.digest'; mode='bad-utc' }
    )

    $prepared = New-Object Collections.ArrayList
    foreach ($definition in $definitions) {
        $runId = [Guid]::NewGuid().ToString('N')
        $obj = New-BaseManifest $seedText $runId $childSha

        switch ([string]$definition.mode) {
            'missing-key' { $obj.PSObject.Properties.Remove('counts') }
            'extra-key' { $obj | Add-Member -NotePropertyName 'unexpectedA25n' -NotePropertyValue $true }
            'wrong-schema' { $obj.schema = 'ait25-provisional-manifest/999' }
            'traversal' { $obj.entries[1].path = '..\escape' }
            'path-case' {
                $d3 = @($obj.entries | Where-Object { $_.path -ceq 'd3' })
                $e = @($obj.entries | Where-Object { $_.path -ceq 'e' })
                if ($d3.Count -ne 1 -or $e.Count -ne 1) {
                    throw 'controlled case-collision entries are not unique'
                }
                $e[0].path = 'D3'
            }
            'root-id' { $obj.provenance.sourceRootId = [string]$obj.entries[1].sourceId }
            'repeat-id' { $obj.entries[1].sourceId = [string]$obj.entries[0].sourceId }
            'policy' { $obj.policy.namedStreams = ([string]$obj.policy.namedStreams) + ' CONTRADICTORY' }
            'bad-utc' { $obj.createdUtc = '/Date(0)/' }
        }

        $text = Convert-ManifestObjectToText $obj
        switch ([string]$definition.mode) {
            'raw-truncate' {
                if (-not $text.EndsWith('}')) { throw 'malformed-json base lacks final object close' }
                $text = $text.Substring(0, $text.Length - 1)
            }
            'dup-exact' { $text = Add-RootDuplicate $text $runId 'exact' }
            'dup-escaped' { $text = Add-RootDuplicate $text $runId 'escaped' }
            'dup-case' { $text = Add-RootDuplicate $text $runId 'case' }
            'dup-stream' { $text = Add-NestedStreamDuplicate $text $obj }
        }

        $bytes = $Utf8.GetBytes($text)
        $actualSha = Get-ShaBytes $bytes
        $launchPin = $actualSha
        if ([string]$definition.mode -ceq 'wrong-pin') {
            $launchPin = Get-WrongDigest $actualSha
        }
        if ($launchPin -cnotmatch '\A[0-9A-F]{64}\z') {
            throw "invalid CLI pin prepared for $($definition.name)"
        }

        [void]$prepared.Add([ordered]@{
            definition = $definition
            runId = $runId
            bytes = $bytes
            actualSha = $actualSha
            launchPin = $launchPin
        })
    }

    $wrong = @($prepared | Where-Object {
        $_.definition.mode -ceq 'wrong-pin'
    })
    if ($wrong.Count -ne 1 -or $wrong[0].launchPin -ceq $wrong[0].actualSha) {
        throw 'wrong-external-digest preparation failed'
    }
    foreach ($item in @($prepared | Where-Object {
        $_.definition.mode -cne 'wrong-pin' -and $_.definition.mode -cne 'omit'
    })) {
        if ($item.launchPin -cne $item.actualSha) {
            throw "malformed case was not repinned: $($item.definition.name)"
        }
        if ($item.launchPin -ceq $wrong[0].launchPin) {
            throw "wrong digest collides with malformed-case pin: $($item.definition.name)"
        }
    }

    $caseNumber = 0
    foreach ($item in $prepared) {
        $caseNumber++
        $definition = $item.definition
        Write-Output ("[a25n] case {0}/{1}: {2} prepare" -f
            $caseNumber, $prepared.Count, $definition.name)

        $caseFailures = New-Object Collections.ArrayList
        $generation = Join-Path $GenerationRoot ('g-' + $item.runId)
        $recovery = Join-Path $generation 'recovery'
        $manifestPath = Join-Path $recovery 'manifest.json'
        $restorePath = Join-Path $generation 'restore'
        $childReportPath = Join-Path $generation 'child-report.json'
        $stdoutPath = Join-Path $generation 'child.stdout.bin'
        $stderrPath = Join-Path $generation 'child.stderr.bin'
        $launch = $null
        $childReport = $null
        $lastStep = $null
        $manifestBefore = $null
        $manifestAfter = $null
        $pinsAfterCase = $null

        try {
            if ([IO.Path]::GetDirectoryName($generation) -cne $GenerationRoot -or
                [IO.Path]::GetFileName($generation) -cnotmatch '\Ag-[0-9a-f]{32}\z') {
                throw 'generation CLI binding is invalid'
            }

            New-BoundDirectory $generation
            New-BoundDirectory $recovery

            if ([string]$definition.mode -cne 'omit') {
                Write-BoundNew $manifestPath $item.bytes
                $manifestBefore = Get-FileSha $manifestPath
                if ($manifestBefore -cne $item.actualSha) {
                    throw 'case manifest write/readback hash mismatch'
                }
            } else {
                if ([Ait25ProvisionalCapture.Native]::Probe($manifestPath) -eq 0) {
                    throw 'missing-manifest case unexpectedly has a manifest'
                }
            }

            Write-Output ("[a25n] case {0}/{1}: {2} launch" -f
                $caseNumber, $prepared.Count, $definition.name)
            $launch = Invoke-Child $ChildPath $generation $item.launchPin

            Write-BoundNew $stdoutPath ([byte[]]$launch.stdoutData)
            Write-BoundNew $stderrPath ([byte[]]$launch.stderrData)

            if ($launch.timedOut) { [void]$caseFailures.Add('child timed out') }
            if ($null -eq $launch.exitCode -or [int]$launch.exitCode -eq 0) {
                [void]$caseFailures.Add("actual exit was not nonzero: $($launch.exitCode)")
            }
            if (-not $launch.stdoutDrained -or -not $launch.stderrDrained) {
                [void]$caseFailures.Add('stdout/stderr drain did not reach EOF')
            }
            if ($launch.stdoutOverflow -or $launch.stderrOverflow) {
                [void]$caseFailures.Add('stdout/stderr exceeded the 1 MiB per-stream bound')
            }
            if ($null -ne $launch.stdoutDrainError -or $null -ne $launch.stderrDrainError) {
                [void]$caseFailures.Add('stdout/stderr drain error')
            }

            if ([Ait25ProvisionalCapture.Native]::Probe($restorePath) -eq 0) {
                [void]$caseFailures.Add('restore output exists')
            }

            if ([Ait25ProvisionalCapture.Native]::Probe($childReportPath) -ne 0) {
                [void]$caseFailures.Add('child-report.json absent')
            } else {
                $reportBytes = Read-Bounded $childReportPath $MaxReportBytes
                $reportText = [Ait25ProvisionalCapture.JsonScan]::Validate(
                    $reportBytes, 32, 2000000
                )
                $childReport = ConvertFrom-Json -InputObject $reportText

                if ([string]$childReport.category -cne [string]$definition.expectedCategory) {
                    [void]$caseFailures.Add(
                        "category '$($childReport.category)' != '$($definition.expectedCategory)'"
                    )
                }
                if ([string]$childReport.message -notmatch [string]$definition.expectedMessage) {
                    [void]$caseFailures.Add(
                        "diagnostic is not attributable to injected fault: $($childReport.message)"
                    )
                }
                if ([bool]$childReport.restoreStarted) {
                    [void]$caseFailures.Add('restoreStarted is not false')
                }
                if ([int]$childReport.exitCode -ne [int]$launch.exitCode) {
                    [void]$caseFailures.Add('report exitCode differs from retained process exit')
                }
                if ([string]$childReport.generation -cne $generation) {
                    [void]$caseFailures.Add('report generation binding mismatch')
                }
                if ([string]$childReport.codeSha256 -cne $childSha) {
                    [void]$caseFailures.Add('report child code pin mismatch')
                }
                if ([string]$childReport.manifestSha256Pinned -cne [string]$item.launchPin) {
                    [void]$caseFailures.Add('report manifest CLI pin mismatch')
                }
                if ([string]$childReport.reportFile -cne $childReportPath -or
                    $null -ne $childReport.reportWriteError) {
                    [void]$caseFailures.Add('bounded child report write binding failed')
                }

                $steps = @($childReport.steps)
                if ($steps.Count -gt 0) { $lastStep = [string]$steps[-1].step }
                if ($lastStep -cne [string]$definition.expectedLastStep) {
                    [void]$caseFailures.Add(
                        "last step '$lastStep' != '$($definition.expectedLastStep)'"
                    )
                }
                if (@($steps | Where-Object {
                    [string]$_.step -ceq 'manifest.valid'
                }).Count -ne 0) {
                    [void]$caseFailures.Add(
                        'child reached manifest.valid instead of refusing the injected fault'
                    )
                }
                if ([string]$childReport.message -match
                    '(?i)recovery top level|recovery\\parent|missing.*parent|path component.*parent') {
                    [void]$caseFailures.Add(
                        'child reached missing recovery/parent instead of the injected fault'
                    )
                }
            }

            if ([string]$definition.mode -cne 'omit') {
                $manifestAfter = Get-FileSha $manifestPath
                if ($manifestAfter -cne $manifestBefore) {
                    [void]$caseFailures.Add('case manifest changed during child execution')
                }
            } elseif ([Ait25ProvisionalCapture.Native]::Probe($manifestPath) -eq 0) {
                [void]$caseFailures.Add('missing manifest was created during child execution')
            }

            if ((Get-FileSha $ChildPath) -cne $childSha) {
                [void]$caseFailures.Add('fresh child code changed during execution')
            }
            $pinsAfterCase = Get-PinSnapshot $pins
            foreach ($pinFailure in @(Compare-PinSnapshot $pinsBefore $pinsAfterCase)) {
                [void]$caseFailures.Add($pinFailure)
            }
        } catch {
            [void]$caseFailures.Add(
                'harness exception: ' + $_.Exception.GetBaseException().Message
            )
        }

        $status = $(if ($caseFailures.Count -eq 0) { 'PASS' } else { 'FAIL' })
        Write-Output ("[a25n] case {0}/{1}: {2} {3}" -f
            $caseNumber, $prepared.Count, $definition.name, $status)

        [void]$Results.Add([ordered]@{
            name = $definition.name
            status = $status
            injectedFault = $definition.mode
            runId = $item.runId
            expectedCategory = $definition.expectedCategory
            expectedMessagePattern = $definition.expectedMessage
            expectedLastStep = $definition.expectedLastStep
            actualExit = $(if ($null -ne $launch) { $launch.exitCode } else { $null })
            actualCategory = $(if ($null -ne $childReport) { $childReport.category } else { $null })
            actualMessage = $(if ($null -ne $childReport) { $childReport.message } else { $null })
            actualLastStep = $lastStep
            restoreStarted = $(if ($null -ne $childReport) { $childReport.restoreStarted } else { $null })
            timedOut = $(if ($null -ne $launch) { $launch.timedOut } else { $null })
            killed = $(if ($null -ne $launch) { $launch.killed } else { $null })
            elapsedMs = $(if ($null -ne $launch) { $launch.elapsedMs } else { $null })
            ownedProcessInspection = $(if ($null -ne $launch) {
                $launch.ownedProcessInspection
            } else { $null })
            noAutomaticRetry = $true
            launchPin = $item.launchPin
            manifestActualSha256 = $item.actualSha
            manifestBefore = $manifestBefore
            manifestAfter = $manifestAfter
            failures = @($caseFailures)
            artifacts = [ordered]@{
                generation = $generation
                manifest = $(if ([string]$definition.mode -ceq 'omit') {
                    'NOT CREATED: injected missing-manifest fault'
                } else { $manifestPath })
                childReport = $childReportPath
                stdout = $stdoutPath
                stderr = $stderrPath
                restore = 'MUST REMAIN ABSENT: ' + $restorePath
            }
            bounded = [ordered]@{
                stdoutBytes = $(if ($null -ne $launch) { $launch.stdoutBytes } else { $null })
                stderrBytes = $(if ($null -ne $launch) { $launch.stderrBytes } else { $null })
                reportBytes = $(if (Test-Path -LiteralPath $childReportPath -PathType Leaf) {
                    (Get-Item -LiteralPath $childReportPath).Length
                } else { $null })
                limitPerArtifactBytes = $MaxReportBytes
            }
        })
    }

    $pinsFinal = Get-PinSnapshot $pins
    $finalPinFailures = @(Compare-PinSnapshot $pinsBefore $pinsFinal)
    if ($finalPinFailures.Count -gt 0) {
        $Fatal = 'external pins changed: ' + ($finalPinFailures -join '; ')
    }
} catch {
    $Fatal = $_.Exception.GetBaseException().Message
}

$requiredNames = @(
    'wrong-external-digest',
    'missing-manifest',
    'malformed-json',
    'duplicate-exact-root-key',
    'duplicate-escaped-root-key',
    'duplicate-case-root-key',
    'nested-stream-duplicate-key',
    'missing-required-key',
    'extra-unknown-key',
    'wrong-schema',
    'traversal-path',
    'case-colliding-entry-path',
    'source-root-id-mismatch',
    'repeated-source-id',
    'contradictory-policy-text',
    'invalid-utc-date-wrapper'
)

$missingResults = @($requiredNames | Where-Object {
    $name = $_
    @($Results | Where-Object { $_.name -ceq $name }).Count -ne 1
})
$passCount = @($Results | Where-Object { $_.status -ceq 'PASS' }).Count
$failCount = @($Results | Where-Object { $_.status -ceq 'FAIL' }).Count
$notRunCount = @($Results | Where-Object { $_.status -ceq 'NOTRUN' }).Count
$allPassed = (
    $null -eq $Fatal -and
    $missingResults.Count -eq 0 -and
    $Results.Count -eq $requiredNames.Count -and
    $passCount -eq $requiredNames.Count
)

$finalReport = [ordered]@{
    schema = 'ait25-manifest-refusal-synthetic/1'
    label = 'SYNTHETIC manifest parsing/semantic refusal tests only'
    synthetic = $true
    allPassed = $allPassed
    exitCode = $(if ($allPassed) { 0 } else { 1 })
    fatal = $Fatal
    fixtureRoot = $FixtureRoot
    reportFile = $ReportPath
    acceptedInputs = [ordered]@{
        compactHarness = $CompactHarness
        frozenCapture = $FrozenCapture
        frozenRestore = $FrozenRestore
        compactReport = $CompactReportPath
        acceptedManifest = $AcceptedManifestPath
        acceptedChild = $AcceptedChildPath
    }
    childAdaptation = [ordered]@{
        freshChild = $(if ($FixtureRoot) { Join-Path $FixtureRoot 'x\r.ps1' } else { $null })
        changedBinding = 'GenerationsRoot exact assignment only'
        oldSyntheticSourcePathLiteralPreserved = $true
        oldSyntheticSourceParentLiteralPreserved = $true
        oldSyntheticForbiddenRootPreserved = $true
        inverseByteProofRequired = $true
        parseErrorsRequired = 0
        staticRootedPathReviewRequired = $true
    }
    summary = [ordered]@{
        required = $requiredNames.Count
        PASS = $passCount
        FAIL = $failCount
        NOTRUN = $notRunCount
        missingOrRepeated = @($missingResults)
    }
    cases = @($Results)
    limits = [ordered]@{
        caseCeilingMs = $CaseCeilingMs
        overallCeilingMs = $OverallCeilingMs
        stdoutBytes = $MaxStreamBytes
        stderrBytes = $MaxStreamBytes
        childReportBytes = $MaxReportBytes
        harnessReportBytes = $MaxReportBytes
        drainWaitMs = $DrainWaitMs
        outerWatchdog = 'SEPARATE AND STILL REQUIRED'
        automaticRetry = 'NONE'
    }
    notRun = [ordered]@{
        positiveRecoveryControl = 'NOT RUN: this first negative phase has no full recovery tree'
        recoveryContentCorruption = 'NOT RUN'
        ordinaryDataCorruption = 'NOT RUN'
        lxCorruption = 'NOT RUN'
        adsCorruption = 'NOT RUN'
        sourceDrift = 'NOT RUN'
        scale = 'NOT RUN'
        originalRuntime = 'NOT RUN; original project/payload paths are provenance strings only'
    }
    safety = [ordered]@{
        createNewOnly = $true
        deletion = 'NONE'
        overwrite = 'NONE'
        merge = 'NONE'
        originalProjectAccess = 'REFUSED'
        credentialsOrOAuth = 'NONE'
        git = 'NONE'
        retainedArtifacts = 'ALL'
    }
    elapsedMs = $Clock.ElapsedMilliseconds
    finishedUtc = [DateTime]::UtcNow.ToString('o')
}

if ($ReportPath -and $FixtureRoot -and
    [Ait25ProvisionalCapture.Native]::Probe($FixtureRoot) -eq 0) {
    try {
        $finalBytes = $Utf8.GetBytes(
            (ConvertTo-Json -Depth 20 -InputObject $finalReport)
        )
        if ($finalBytes.Length -gt $MaxReportBytes) {
            throw "final report exceeds $MaxReportBytes bytes"
        }
        Write-BoundNew $ReportPath $finalBytes
        $ReportWritten = $true
        Write-Output "[a25n] report: $ReportPath"
    } catch {
        Write-Error ("final CREATE_NEW report write failed: " +
            $_.Exception.GetBaseException().Message)
    }
}

if (-not $ReportWritten -or -not $allPassed) { exit 1 }
exit 0
