```python
\# scripts/prospecting/manager/workflows.py
from __future__ import annotations
from dataclasses import dataclass
import json
from pathlib import Path
import re
from scripts.prospecting.manager.p5_contracts import ENTRYPOINTS

AGENTS = {"prospecting-list-builder", "prospecting-personalizer", "prospecting-campaigner", "inspector", "human"}
AGENT_CLI = {"prospecting-list-builder":"list-builder", "prospecting-personalizer":"personalizer",
             "prospecting-campaigner":"campaigner"}
LOCAL_OPERATIONS = {"inspector":{"grade"}, "human":{"review"}}
SLUG = re.compile(r"^[a-z0-9]+(?:-[a-z0-9]+)*$")
@dataclass(frozen=True)
class Stage: id: str; agent: str; operation: str; needs: tuple[str, ...]; inspect: bool
@dataclass(frozen=True)
class Workflow: id: str; manager: str; version: int; stages: tuple[Stage, ...]

def load_workflow(path: Path) -> Workflow:
    text = path.read_text(encoding="utf-8")
    if not text.startswith("---\n") or "\n---\n" not in text[4:]: raise ValueError("workflow_frontmatter_required")
    raw = text.split("\n---\n", 1)[0][4:]
    values = json.loads(raw)
    if set(values) != {"id","default-manager","version","stages"}: raise ValueError("invalid_workflow_shape")
    if not SLUG.fullmatch(values["id"]) or type(values["version"]) is not int or values["version"] < 1: raise ValueError("invalid_workflow_identity")
    if values["default-manager"] != "prospecting-manager": raise ValueError("invalid_manager")
    seen: set[str] = set(); stages = []
    for item in values["stages"]:
        if set(item) != {"id","agent","operation","needs","inspect"}: raise ValueError("invalid_stage_shape")
        if item["id"] in seen or not SLUG.fullmatch(item["id"]) or item["agent"] not in AGENTS or not set(item["needs"]).issubset(seen): raise ValueError("invalid_stage_graph")
        if type(item["inspect"]) is not bool: raise ValueError("inspect_must_be_boolean")
        allowed = (set(ENTRYPOINTS[AGENT_CLI[item["agent"]]]["commands"])
                   if item["agent"] in AGENT_CLI else LOCAL_OPERATIONS[item["agent"]])
        if item["operation"] not in allowed: raise ValueError("unknown_operation")
        seen.add(item["id"]); stages.append(Stage(item["id"], item["agent"], item["operation"], tuple(item["needs"]), item["inspect"]))
    return Workflow(values["id"], values["default-manager"], int(values["version"]), tuple(stages))
```
