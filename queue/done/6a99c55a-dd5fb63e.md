\## Architecture

P4 adds a campaigner package behind P1's SQLite repository and deterministic executor. The campaigner owns enrollment, delivery scheduling, inbound classification, suppression, labels, reply revisions, and draft release. It has no Gmail credential or raw Gmail tool. All four Gmail capabilities are typed `exec_request` operations handled by `gmail_adapter.py`; the adapter talks to the installed Gmail MCP only outside `KB_PROSPECTING_NO_NETWORK=1`, and otherwise uses a stateful fake/recorded transport.

Release is a bounded state machine. For every due delivery the executor serializes the Gmail thread, refreshes it, rechecks all stops, blocks, approval hashes, caps, and tier, then CAS-claims the delivery. That refresh plus CAS is the linearization point. P4 creates a draft and persists its deterministic RFC Message-ID and Gmail identifiers. An inbound arriving after the CAS is outside the prevention guarantee and is reconciled on the next inbound pass.

```python
"""Counts-only campaigner sweep, scan, and status commands."""
from __future__ import annotations
import argparse
import json
import sqlite3
from dataclasses import dataclass
from datetime import datetime, timezone
from typing import Callable, Iterable
from scripts.prospecting import store
from scripts.prospecting.campaigner.inbound import InboundEnvelope, process_inbound
from scripts.prospecting.campaigner.release import ReleaseResult
from scripts.prospecting.executor_campaigner import build_campaigner_service


@dataclass
class CampaignerService:
    connection: sqlite3.Connection
    inbound_source: Callable[[], Iterable[InboundEnvelope]]
    release: Callable[[str], ReleaseResult]
    now: Callable[[], str] = lambda: datetime.now(timezone.utc).isoformat()
    breaker: Callable[[], bool] = lambda: False

    def scan(self) -> dict[str, int]:
        total = {'processed': 0, 'stopped': 0, 'blocked': 0}
        for envelope in self.inbound_source():
            item = process_inbound(self.connection, envelope).as_dict()
            for key in total:
                total[key] += item[key]
        return total

    def sweep(self) -> dict[str, int]:
        scanned = self.scan()
        result = {**scanned, 'considered': 0, 'drafted': 0, 'cancelled': 0,
                  'cap_blocked': 0, 'warning_paused': 0, 'next_due_count': 0}
        if self.breaker():
            result['warning_paused'] = 1
            result['next_due_count'] = self.connection.execute(
                "SELECT count(*) FROM delivery WHERE state='reserved'"
            ).fetchone()[0]
            return result
        rows = self.connection.execute(
            "SELECT delivery_id FROM delivery WHERE state='reserved' AND scheduled_at<=? ORDER BY scheduled_at,delivery_id",
            (self.now(),),
        ).fetchall()
        for (delivery_id,) in rows:
            result['considered'] += 1
            state = self.release(delivery_id).state
            if state in result:
                result[state] += 1
        second = self.scan()
        for key in ('processed', 'stopped', 'blocked'):
            result[key] += second[key]
        result['next_due_count'] = self.connection.execute(
            "SELECT count(*) FROM delivery WHERE state='reserved'"
        ).fetchone()[0]
        return result

    def status(self) -> dict[str, int]:
        due = self.connection.execute("SELECT count(*) FROM delivery WHERE state='reserved'").fetchone()[0]
        paused = self.connection.execute("SELECT count(*) FROM enrollment WHERE status='blocked'").fetchone()[0]
        replied = self.connection.execute("SELECT count(*) FROM enrollment WHERE stop_reason='human_reply'").fetchone()[0]
        return {'due': due, 'paused': paused, 'replied': replied}


def run(service: CampaignerService, command: str) -> dict[str, int]:
    return {'scan': service.scan, 'sweep': service.sweep, 'status': service.status}[command]()


def main(
    argv: list[str] | None = None,
    service: CampaignerService | None = None,
    open_store_fn: Callable[[], sqlite3.Connection] | None = None,
    service_factory: Callable[[sqlite3.Connection], CampaignerService] | None = None,
    emit: Callable[[str], None] = print,
) -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument('command', choices=('sweep', 'scan', 'status'))
    args = parser.parse_args(argv)
    if service is None:
        connection = (open_store_fn or store.open_store)()
        service = (service_factory or build_campaigner_service)(connection)
    emit(json.dumps(run(service, args.command), sort_keys=True, separators=(',', ':')))
    return 0
```
