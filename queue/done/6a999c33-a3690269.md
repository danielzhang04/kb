from dataclasses import dataclass
import os, random
from pathlib import Path
from urllib.parse import urlsplit

@dataclass(frozen=True)
class BrowserPolicy: allowed_hosts: tuple[str,...]; rolling_cap: int=40; session_seconds: int=1800
def check_navigation(url,policy):
    parsed=urlsplit(url)
    if parsed.scheme=="file" and os.environ.get("KB_PROSPECTING_NO_NETWORK")=="1": return
    if parsed.scheme!="https" or parsed.hostname not in policy.allowed_hosts: raise ValueError("browser_domain_blocked")
    if parsed.scheme=="https" and not parsed.path.startswith(("/search/results/people/","/in/","/company/")): raise ValueError("browser_action_blocked")
def next_delay(random_source): return random_source.randint(45,120)
def check_action(action):
    allowed={"search_results","profile_page","company_page","close","wait"}
    if action not in allowed: raise ValueError(f"browser_action_blocked:{action}")
def detect_stop(html):
    lowered=html.casefold()
    return next((m.replace(" ","_") for m in ("captcha","checkpoint","rate warning","login challenge","unexpected modal") if m in lowered),None)

def open_persistent_page(playwright,user_data_dir):
    context=playwright.chromium.launch_persistent_context(str(user_data_dir),channel="chrome",headless=False)
    return context.pages[0] if context.pages else context.new_page()

def dedicated_user_data_dir(environ=None):
    values=os.environ if environ is None else environ
    root=values.get("LOCALAPPDATA")
    if not root: raise RuntimeError("localappdata_unavailable")
    return Path(root)/"kb-outreach-chrome"

class _Fields(HTMLParser):
    def __init__(self): super().__init__(); self.values={}; self.current=None
    def handle_starttag(self,tag,attrs):
        data=dict(attrs); self.current=data.get("data-field")

        cutoff=(self.clock.now()-timedelta(hours=24)).isoformat()
        return self.connection.execute("SELECT COUNT(*) FROM audit WHERE action='linkedin_profile_load_reserved' AND at>?",(cutoff,)).fetchone()[0]>=40
