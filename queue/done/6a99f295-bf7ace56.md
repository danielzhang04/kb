\## Architecture

The VM may assemble IDs and display a PII-free summary, but it never receives names, addresses,
message bytes, credentials, or a send capability. The canonical approval scope is staged as the
work-order bytes of a normal kb approval card. The dashboard signs those committed bytes; the
desktop verifier imports the existing kb WebAuthn verifier, acts only on the verifier's pinned
card/result, and never parses public-key material itself.

A batch scope binds the campaign, policy, sorted exact revision/contact pairs, batch hash, mailbox,
T1 tier, permitted window, expiry, and a random parent nonce. Materialization is one transaction.
The first approval uses the parent nonce and later rows use deterministic child nonces derived from
the parent nonce plus the exact item. This preserves the P1 unique approval.nonce contract, makes
replaying the assertion fail the entire transaction, and leaves each send with a separately
consumable nonce. Each row's scope_hash is the canonical hash of its exact send scope.

The T1 executor is the only live-send path. It scans inbound first, refreshes the Gmail thread,
rechecks every bound field and all hooks, compare-and-set claims the delivery, and consumes the
matching approval in the same SQLite transaction immediately before the Gmail adapter call.
Follow-ups continue through P4's draft path. T2, T3, sales, and reply-template automation remain
fail-closed.

```python
\# scripts/prospecting/approval/verify.py
from __future__ import annotations
from datetime import datetime
import hashlib
import sqlite3
import uuid
from pathlib import Path

import scripts.approvals as approvals
from scripts.webauthn_verify import verify_webauthn_approval
from scripts.prospecting.approval.scope import (
    deserialize, materialized_send_scopes,
)

def _human(value: str) -> bool:
    return bool(value) and not value.endswith("@agents.local") and value not in {
        "agent","executor","prospecting-campaigner","codex-worker",
    }

def verify_and_insert(
    connection: sqlite3.Connection,
    assertion_ref: str,
    approved_card_path: Path,
    credentials_path: Path,
    expected_scope_hash: str,
    now: datetime,
    manage_transaction: bool = True,
) -> tuple[str, ...]:
    result = verify_webauthn_approval(
        approval_path=approved_card_path,
        assertion_ref=assertion_ref,
        credential_store_path=credentials_path,
        now=now,
    )
    if not getattr(result, "ok", False) or result.card is None:
        raise ValueError("webauthn_rejected")
    card = result.card
    if card.meta.get("risk-tier") != "T3" or card.meta.get("action") != "approve-prospecting-send":
        raise ValueError("webauthn_scope_action")
    approver = str(card.meta.get("owner",""))
    if not _human(approver):
        raise ValueError("human_approver_required")
    scope_text = approvals.work_order_of(card.body).strip()
    scope = deserialize(scope_text, expected_scope_hash, now)
    rows = materialized_send_scopes(scope)
    ids: list[str] = []
    try:
        if manage_transaction:
            connection.execute("BEGIN IMMEDIATE")
        if connection.execute(
            "SELECT 1 FROM approval WHERE nonce=?", (scope.nonce,)
        ).fetchone():
            raise ValueError("approval_replay")
        campaign = connection.execute(
            "SELECT policy_hash,approval_tier FROM campaign WHERE campaign_id=?",
            (scope.campaign_id,),
        ).fetchone()
        if campaign != (scope.policy_hash, "T1"):
            raise ValueError("scope_mismatch")
        for row in rows:
            match = connection.execute(
                "SELECT 1 FROM revision r JOIN contact_point c ON c.person_id=r.person_id "
                "WHERE r.hash=? AND r.campaign_id=? AND c.contact_id=? AND c.state='valid'",
                (row["revision_hash"],scope.campaign_id,row["contact_id"]),
            ).fetchone()
            if match is None:
                raise ValueError("scope_mismatch")
            approval_id = str(uuid.uuid4())
            connection.execute(
                "INSERT INTO approval VALUES(?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)",
                (
                    approval_id,assertion_ref,row["campaign_id"],row["policy_hash"],
                    row["content_kind"],row["revision_hash"],row["contact_id"],
                    row["mailbox_id"],approver,now.isoformat(),row["expires_at"],
                    row["tier"],row["send_window"],row["nonce"],
                    row["permitted_action"],None,row["scope_hash"],None,
                ),
            )
            ids.append(approval_id)
        if manage_transaction:
            connection.commit()
    except Exception:
        if manage_transaction:
            connection.rollback()
        raise
    return tuple(ids)
```
