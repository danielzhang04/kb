def metadata_snapshot(helper, root, tree):
    """File IDs/basic stat/attributes only. No owner, DACL, SACL, or full SD claim."""
    snapshot = {}
    for rel in tree:
        full = root if rel == "." else os.path.join(root, rel)
        stat = os.stat(full, follow_symlinks=False)
        snapshot[rel] = {
            "fileId": [stat.st_dev, stat.st_ino],
            "attributes": helper.attrs(full),
            "links": stat.st_nlink,
            "size": stat.st_size,
            "creationOrCtimeNs": stat.st_ctime_ns,
            "writeTimeNs": stat.st_mtime_ns,
        }
    return snapshot

def run_routes(helper, fixture, writer, project):

    results = []
    compiled = False
    aborted = None

    for route in build_routes(helper, fixture, writer, project):
        rid = route["id"]
        if aborted or (route["needsCompile"] and not compiled):
            results.append({
                "id": rid,
                "status": "FAIL",
                "classification": "not-run",
                "detail": aborted or "J0-compile did not pass",
            })
            continue

        try:
            project_before, metadata_before = project_snapshot(helper, project)
            fixture_before = helper.fixture_walk(fixture)

            for path in [route["cwd"], *route["paths"]]:
                helper.need(
                    helper.under(path, project) or helper.under(path, fixture),
                    f"route path outside destination/fixture: {path}",
                )
                helper.exact(path)
            helper.check_chain(route["cwd"])

            record = helper.run_process(writer, logs, route)
            record["classification"], record["status"], record["detail"] = (
                helper.classify(record, route)
            )

            stdout = helper.read_file(record["stdoutLog"], helper.MAX_STREAM)
            stderr = helper.read_file(record["stderrLog"], helper.MAX_STREAM)
            old_root_hits = (
                stdout.lower().count(OLD_ROOT_BYTES)
                + stderr.lower().count(OLD_ROOT_BYTES)
            )
            record["oldRootDiagnostic"] = {
                "occurrences": old_root_hits,
                "interpretation": (
                    "diagnostic only; relevance requires route-specific review"
                    if old_root_hits
                    else "not observed"
                ),
            }

            project_after, metadata_after = project_snapshot(helper, project)
            tree_drift = helper.tree_diff(project_before, project_after)
            metadata_drift = sorted(
                rel for rel in set(metadata_before) | set(metadata_after)
                if metadata_before.get(rel) != metadata_after.get(rel)
            )
            record["projectContentDrift"] = tree_drift[:50]
            record["projectMetadataDrift"] = metadata_drift[:50]
            record["sourceBytecodeCreated"] = any(
                rel.casefold().endswith((".pyc", ".pyo")) for rel in tree_drift
            )

            fixture_after = helper.fixture_walk(fixture)
            fixture_changed = [
                rel for rel in sorted(set(fixture_before) | set(fixture_after))
                if fixture_before.get(rel) != fixture_after.get(rel)
            ]
            bad_fixture_changes = [
                rel for rel in fixture_changed if not helper.is_rt(rel)
            ]
            record["fixtureChangedPaths"] = fixture_changed[:50]

            if tree_drift or metadata_drift or bad_fixture_changes:
                prior = record["detail"]
                record["classification"] = "side-effect-failure"
                record["status"] = "FAIL"
                record["detail"] = (
                    f"destination drift={tree_drift[:20]}, "
                    f"metadata drift={metadata_drift[:20]}, "
                    f"fixture changes outside _rt={bad_fixture_changes[:20]}; "
                    f"earlier result: {prior}"
                )
                aborted = f"side effect in {rid}"
            elif record["classification"] == "unsafe-process-state":
                aborted = f"unsafe direct-child state in {rid}"

        except (helper.Fail, OSError, ValueError) as error:
            record = {
                "id": rid,
                "status": "FAIL",
                "classification": "harness-refused",
                "detail": f"{type(error).__name__}: {error}",
            }
            aborted = f"harness refusal in {rid}"

        if rid == "J0-compile":
            compiled = record["status"] == "PASS"
        results.append(record)

    results.append({
        "id": "hw02-editor-launch",
        "status": "HOLD",
        "classification": "pre-existing-editor-defect",
        "detail": "StartupFile target was not launched or repaired",
    })
    return results


def main():
    parser = argparse.ArgumentParser(
        description="Candidate destination-runtime proof; no move or approval"
    )
    modes = parser.add_mutually_exclusive_group(required=True)
    modes.add_argument("--rehearse", action="store_true")
    modes.add_argument("--live", action="store_true")
    parser.add_argument("--generation", required=True)
    parser.add_argument("--expected-manifest-sha256", required=True)
    parser.add_argument("--confirm-window-run-id")
    parser.add_argument("--accepted-mover-run-id")
    parser.add_argument("--outer-watchdog-seconds", type=int, required=True)
    args = parser.parse_args()

    mode = "live" if args.live else "rehearse"
    if not HEX64.fullmatch(args.expected_manifest_sha256):
        parser.error("--expected-manifest-sha256 must be 64 uppercase hex")
    if args.outer_watchdog_seconds != 600:
        parser.error("an external 600-second process-tree watchdog is required")
    if mode == "live":
        if not args.confirm_window_run_id or not args.accepted_mover_run_id:
            parser.error(
                "live mode requires --confirm-window-run-id and "
                "--accepted-mover-run-id; these are context assertions, not approval"
            )
        if not HEX32.fullmatch(args.confirm_window_run_id):
            parser.error("--confirm-window-run-id must be 32 lowercase hex")
        if not (1 <= len(args.accepted_mover_run_id) <= 128):
            parser.error("--accepted-mover-run-id has invalid length")
    elif args.confirm_window_run_id or args.accepted_mover_run_id:
        parser.error("live confirmation arguments are forbidden in rehearse mode")

    fixture = None
    try:
        helper = load_helper()
        run_id = generation_run_id(args.generation)
        if mode == "live" and run_id != args.confirm_window_run_id:
            raise helper.Fail("window confirmation does not match generation run ID")

        ctx = helper.preflight(args.generation)
        helper.need(
            ctx["pin"] == args.expected_manifest_sha256,
            "verified generation pin differs from explicit trusted pin",
        )
        verify_reviewed_execution_view(helper, ctx["view"])

        expected_destination_view = dict(ctx["view"])
        overlay = {}
        if mode == "live":
            manifest = helper.parse_json(
                helper.read_file(
                    os.path.join(args.generation, "recovery", "manifest.json"),
                    helper.MAX_JSON,
                ),
                os.path.join(args.generation, "recovery", "manifest.json"),
            )
            provenance = manifest.get("provenance")
            helper.need(
                isinstance(provenance, dict)
                and provenance.get("sourcePath") == LIVE_SOURCE
                and provenance.get("sourceParentPath") == LIVE_SOURCE_PARENT,
                "window capture is not bound to the fixed live source",
            )
            assert_absent_child(helper, LIVE_SOURCE_PARENT, "ai")
            helper.check_chain(helper.exact(LIVE_DESTINATION))
            overlay = load_preview(helper, ctx["view"])
            expected_destination_view.update(overlay)
            actual = helper.walk(LIVE_DESTINATION)
            differences = helper.tree_diff(expected_destination_view, actual)
            helper.need(
                not differences,
                "live destination does not equal window capture plus five "
                f"preview postimages: {differences[:20]}",
            )

        fixture, writer, project = make_fixture(helper, mode, ctx)
        if mode == "rehearse":
            copied = helper.walk(project)
            differences = helper.tree_diff(ctx["view"], copied)
            helper.need(
                not differences,
                f"isolated destination copy differs from generation: {differences[:20]}",
            )

        initial_tree, initial_metadata = project_snapshot(helper, project)
        results = run_routes(helper, fixture, writer, project)
        final_tree, final_metadata = project_snapshot(helper, project)
        final_drift = helper.tree_diff(initial_tree, final_tree)
        final_metadata_drift = sorted(
            rel for rel in set(initial_metadata) | set(final_metadata)
            if initial_metadata.get(rel) != final_metadata.get(rel)
        )
        helper.verify_generation(ctx)
        helper.need(
            digest(raw_read(HELPER_PATH, MAX_HELPER)) == HELPER_SHA256,
            "helper changed across run",
        )

        statuses = [row["status"] for row in results]
        checks = [
            {
                "name": "project.unchangedAcrossRun",
                "status": "PASS" if not final_drift else "FAIL",
                "detail": final_drift[:50] or "entry set, lengths and SHA256 unchanged",
            },
            {
                "name": "project.basicMetadataUnchangedAcrossRun",
                "status": "PASS" if not final_metadata_drift else "FAIL",
                "detail": final_metadata_drift[:50] or (
                    "file IDs, attributes, links, sizes, creation/ctime and write times unchanged"
                ),
            },
            {
                "name": "generation.unchangedAfterRun",
                "status": "PASS",
                "detail": "helper generation verifier completed after all routes",
            },
        ]
        statuses.extend(row["status"] for row in checks)
        summary = {
            status: statuses.count(status)
            for status in ("PASS", "EXPECTED", "HOLD", "FAIL")
        }
        exit_code = 1 if summary["FAIL"] else 0

        report = {
            "schema": SCHEMA,
            "runId": os.path.basename(fixture).removeprefix(
                "ai-destination-runtime-"
            ),
            "mode": mode,
            "proofScope": (
                "synthetic destination rehearsal; live old path was not inspected"
                if mode == "rehearse"
                else "live destination post-move runtime evidence; not human approval"
            ),
            "fixture": fixture,
            "projectExecuted": project,
            "generation": args.generation,
            "generationRunId": run_id,
            "manifestSha256Pin": ctx["pin"],
            "previewManifestSha256": (
                PREVIEW_MANIFEST_SHA256 if mode == "live" else None
            ),
            "previewOverlayCount": len(overlay),
            "acceptedMoverRunIdCallerAssertion": (
                args.accepted_mover_run_id if mode == "live" else None
            ),
            "externalWatchdogSecondsCallerAssertion": args.outer_watchdog_seconds,
            "routes": results,
            "checks": checks,
            "summary": summary,
            "status": (
                "destination-runtime-no-unexpected-results"
                if exit_code == 0
                else "destination-runtime-failed-fixture-retained"
            ),
            "claims": [
                "No move, deletion, repair, package installation, Git operation, "
                "environment change, or live-project write is performed.",
                "Live context arguments are assertions only and do not prove human approval.",
                "Homework-04 output is written only in the fresh fixture.",
            ],
            "limitations": [
                "Project content, entry set, file IDs, basic attributes and selected "
                "timestamps are checked; owner, DACL, SACL and full security descriptors "
                "are not read or claimed. The independent live adapter owns full-SD postflight.",
                "Python/JDK and ambient trusted packages remain trusted as found.",
                "The helper kills only direct children. The required external watchdog "
                "must enforce the process-tree terminal bound.",
                "No-follow safety remains check-then-use because Python stdlib cannot "
                "atomically open every Windows path component without following reparses.",
                "Old-root text is diagnostic only; relevance depends on the emitting route.",
                "The adapter consumes existing generation/preview formats and defines no "
                "schema for the independently built live mover.",
            ],
            "finishedUtc": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
            "exitCode": exit_code,
        }
        report_path = os.path.join(fixture, "_rt", "report.json")
        writer.write_new(report_path, json.dumps(report, indent=2).encode("utf-8"))
        print(
            f'status={report["status"]} exit={exit_code} summary={summary} '
            f"fixture={fixture} report={report_path}"
        )
        return exit_code

    except Exception as error:
        return refuse(
            f"{type(error).__name__}: {error}",
            fixture_created=fixture is not None,
        )


if __name__ == "__main__":
    raise SystemExit(main())

Usage: materialize as `scripts/ai-destination-runtime.py`, review it, then invoke with Windows CPython 3.12 using `-I -B`. Both modes require the exact generation, a trusted uppercase manifest pin, and an externally enforced 600-second process-tree watchdog. Live mode additionally requires the matching window run ID and accepted mover run ID.

Limitations: this candidate was not executed or syntax-tested. It does not authorize a move, prove human approval, inspect ACLs/SACLs, or replace the independent live adapter’s full security-descriptor postflight. Rehearsal never probes the real old source path; live mode fails closed unless that fixed path is absent and the destination equals the verified capture plus all five pinned editor postimages.
