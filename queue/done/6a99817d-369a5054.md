from __future__ import annotations
from dataclasses import dataclass
import hashlib, json, os, sqlite3, uuid
from typing import Mapping, Protocol
from scripts.prospecting.store import CreditReservation, ProviderAttempt, insert_provider_attempt, release_credit, reserve_credit, settle_credit

@dataclass(frozen=True)
class VendorResult:
    state: str; credits: int; records: tuple[Mapping[str, object], ...]; raw_response_ref: int | None = None

class VendorAdapter(Protocol):
    def max_cost(self, operation: str) -> int: ...
    def call(self, operation: str, payload: Mapping[str, object], idempotency_key: str) -> VendorResult: ...

def _execute_reserved(connection: sqlite3.Connection, campaign_id: str, exec_request_id: str, person_id: str, provider: str, operation: str, payload: Mapping[str, object], idempotency_key: str, adapter: VendorAdapter, now: str = "2026-09-03T00:00:00Z", monthly_limit: tuple[str,int] | None = None) -> VendorResult:
    \# Private executor hook only. BEGIN IMMEDIATE makes campaign and optional monthly
    \# budget inspection plus reservation one serialization point across connections.
    reservation_id = str(uuid.uuid4()); max_cost = adapter.max_cost(operation)
    connection.execute("BEGIN IMMEDIATE")
    if monthly_limit is not None:
        month,limit=monthly_limit
        used=connection.execute("SELECT COALESCE(SUM(max_cost),0) FROM credit_reservation WHERE provider=? AND substr(created_at,1,7)=? AND state IN ('reserved','settled')",(provider,month)).fetchone()[0]
        if int(used)+max_cost > limit:
            connection.rollback()
            result=VendorResult("skipped_budget",0,())
            _record_attempt(connection,exec_request_id,person_id,provider,operation,payload,result,now)
            return result
    reservation = CreditReservation(reservation_id=reservation_id,campaign_id=campaign_id,provider=provider,
        exec_request_id=exec_request_id,max_cost=max_cost,actual_cost=0,state="reserved",
        created_at=now,settled_at=None)
    if not reserve_credit(connection, reservation):
        connection.rollback()
        result = VendorResult("skipped_budget",0,())
    else:
        connection.commit()
        result=None
        for attempt_number in range(adapter.max_retries+1):
            try:
                result = adapter.call(operation,payload,idempotency_key)
                break
            except TimeoutError:
                if attempt_number==adapter.max_retries: result=VendorResult("error",0,())
            except Exception:
                result=VendorResult("error",0,()); break
        if result.credits: settle_credit(connection,reservation_id,result.credits,now)
        else: release_credit(connection,reservation_id,now)
    _record_attempt(connection,exec_request_id,person_id,provider,operation,payload,result,now)
    return result

def _record_attempt(connection,exec_request_id,person_id,provider,operation,payload,result,now):
    attempt = ProviderAttempt(attempt_id=exec_request_id,person_id=person_id,provider=provider,call=operation,
        input_hash=hashlib.sha256(json.dumps(payload,sort_keys=True).encode()).hexdigest(),priority=0,
        credits=result.credits,result=result.state,started_at=now,finished_at=now,
        raw_response_ref=result.raw_response_ref)
    insert_provider_attempt(connection,attempt)

def execute_vendor_request(executor, request, *, campaign_id, person_id, adapter, now):
    allowed={"vendor.pdl.person_search":"pdl","vendor.apify.profile_batch":"apify",
        "vendor.hunter.find":"hunter","vendor.hunter.verify":"hunter",
        "vendor.snov.find":"snov","vendor.snov.verify":"snov"}
    provider=allowed.get(request.operation)
    if provider is None or provider != adapter.provider:
        raise ValueError("unknown_vendor_operation")
    payload=json.loads(request.payload)
    if _contains_forbidden(payload):
        raise ValueError("forbidden_vendor_payload")
    operation=request.operation.rsplit(".",1)[-1]
    monthly=(now[:7],100) if provider=="pdl" else None
    return _execute_reserved(executor.connection,campaign_id,request.request_id,person_id,
        provider,operation,payload,request.request_id,adapter,now,monthly)

def _contains_forbidden(value):
    forbidden={"cookie","cookies","session","credential","credentials","token","password","li_at"}
    if isinstance(value,Mapping):
        return any(str(key).casefold() in forbidden or _contains_forbidden(item) for key,item in value.items())
    if isinstance(value,(list,tuple)):
        return any(_contains_forbidden(item) for item in value)
    return False

from __future__ import annotations
import hashlib, json, os, uuid
from pathlib import Path
from typing import Callable, Mapping
from scripts.prospecting.providers.base import VendorResult, _execute_reserved
from scripts.prospecting.store import ProviderAttempt, insert_provider_attempt

class PDLSpotAdapter:
    provider="pdl"
    timeout_seconds=20
    max_retries=2
    def __init__(self, fixture_dir: Path = Path("orgs/prospecting/fixtures/vendor/pdl"), transport: Callable[...,Mapping[str,object]] | None = None, clock: Callable[[],float] | None = None, sleeper: Callable[[float],None] | None = None):
        import time
        self.fixture_dir,self.transport=fixture_dir,transport; self.clock=clock or time.monotonic; self.sleeper=sleeper or time.sleep; self.last_call=None; self._results={}
    def max_cost(self, operation: str) -> int:
        if operation != "person_search": raise ValueError("pdl_unknown_operation")
        return 1
    def call(self, operation: str, payload: Mapping[str,object], idempotency_key: str) -> VendorResult:
        if idempotency_key in self._results: return self._results[idempotency_key]
        if os.environ.get("KB_PROSPECTING_NO_NETWORK") == "1":
            path = self.fixture_dir / ("spot-not-found.json" if payload.get("force_not_found") else "spot-success.json")
            raw = json.loads(path.read_text(encoding="utf-8"))
        else:
            if self.transport is None: raise RuntimeError("pdl_transport_unavailable")
            now=self.clock()
            if self.last_call is not None: self.sleeper(max(0.0,1.0-(now-self.last_call)))
            self.last_call=now
            raw = self.transport(operation,payload,idempotency_key,self.timeout_seconds)
        result=VendorResult(str(raw["result"]),int(raw["credits"]),tuple(raw.get("records",())))
        self._results[idempotency_key]=result; return result

def remaining_pdl_free_tier(connection, month: str) -> int:
    used = connection.execute("SELECT COALESCE(SUM(max_cost),0) FROM credit_reservation WHERE provider='pdl' AND substr(created_at,1,7)=? AND state IN ('reserved','settled')",(month,)).fetchone()[0]
    return max(0,100-int(used))

def execute_pdl_spot(connection,campaign_id,exec_request_id,person_id,payload,idempotency_key,month,adapter,now="2026-09-03T00:00:00Z"):
    return _execute_reserved(connection,campaign_id,exec_request_id,person_id,"pdl","person_search",
        payload,idempotency_key,adapter,now,(month,100))

            input_hash TEXT NOT NULL,
            priority INTEGER NOT NULL,
            credits INTEGER NOT NULL CHECK(credits>=0),
            result TEXT NOT NULL,

        connection.execute("ALTER TABLE provider_attempt_p2 RENAME TO provider_attempt")
